import {
  DialogueSegment,
  CharacterProfile,
  WhisperDisagreement,
  ProductionLog,
  DialogueStyle,
  LipSyncMode,
  AudioMixMode
} from '../types';

const DB_NAME = 'AngkorDubProductionDB_V2';
const DB_VERSION = 1;
const STORE_NAME = 'projects';
const CURRENT_PROJECT_KEY = 'active_production_project';

export interface SavedProjectCheckpoint {
  projectName: string;
  updatedAt: string;
  sourceFileName: string | null;
  sourceVideoDuration: number;
  sourceLanguage: 'en' | 'km';
  targetLanguage: 'km' | 'en';
  dialogueStyle: DialogueStyle;
  lipSyncMode: LipSyncMode;
  audioMixMode: AudioMixMode;
  rawScribeResponseJson: string | null;
  segments: DialogueSegment[];
  characters: CharacterProfile[];
  disagreements: WhisperDisagreement[];
  logs: ProductionLog[];
}

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);
    request.onupgradeneeded = () => {
      const db = request.result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME);
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}

export async function saveCheckpointToIndexedDB(checkpoint: SavedProjectCheckpoint): Promise<void> {
  const db = await openDatabase();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite');
    const store = tx.objectStore(STORE_NAME);
    store.put(checkpoint, CURRENT_PROJECT_KEY);
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}

export async function loadCheckpointFromIndexedDB(): Promise<SavedProjectCheckpoint | null> {
  try {
    const db = await openDatabase();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readonly');
      const store = tx.objectStore(STORE_NAME);
      const request = store.get(CURRENT_PROJECT_KEY);
      request.onsuccess = () => resolve(request.result || null);
      request.onerror = () => reject(request.error);
    });
  } catch {
    return null;
  }
}

export async function clearCheckpointInIndexedDB(): Promise<void> {
  const db = await openDatabase();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite');
    const store = tx.objectStore(STORE_NAME);
    store.delete(CURRENT_PROJECT_KEY);
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}
