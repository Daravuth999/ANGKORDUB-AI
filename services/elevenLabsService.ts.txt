/**
 * Extracts a compact 16kHz Mono WAV file from a video or audio File in the browser.
 * This prevents Scribe v2 / Whisper upload timeouts and container codec errors on large movie files.
 */
export async function extractCompactAudioWavFromFile(
  mediaFile: File,
  onProgress?: (msg: string) => void
): Promise<File> {
  if (mediaFile.type === 'audio/wav' && mediaFile.size < 15 * 1024 * 1024) {
    return mediaFile;
  }

  onProgress?.('Extracting 16kHz mono PCM audio track from media container for reliable transcription...');
  const arrayBuffer = await mediaFile.arrayBuffer();
  const audioCtx = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();

  try {
    const decoded = await audioCtx.decodeAudioData(arrayBuffer.slice(0));
    const targetSampleRate = 16000;
    const totalSamples = Math.ceil(decoded.duration * targetSampleRate);
    const offlineCtx = new OfflineAudioContext(1, totalSamples, targetSampleRate);

    const source = offlineCtx.createBufferSource();
    source.buffer = decoded;
    source.connect(offlineCtx.destination);
    source.start(0);

    const rendered = await offlineCtx.startRendering();
    const wavBlob = monoAudioBufferToWavBlob(rendered);
    const cleanName = mediaFile.name.replace(/\.[^/.]+$/, '') + '_16khz.wav';
    return new File([wavBlob], cleanName, { type: 'audio/wav' });
  } catch {
    return mediaFile;
  } finally {
    await audioCtx.close();
  }
}

function monoAudioBufferToWavBlob(buffer: AudioBuffer): Blob {
  const numSamples = buffer.length;
  const out = new ArrayBuffer(44 + numSamples * 2);
  const view = new DataView(out);
  const channelData = buffer.getChannelData(0);

  const writeStr = (offset: number, str: string) => {
    for (let i = 0; i < str.length; i++) {
      view.setUint8(offset + i, str.charCodeAt(i));
    }
  };

  writeStr(0, 'RIFF');
  view.setUint32(4, 36 + numSamples * 2, true);
  writeStr(8, 'WAVE');
  writeStr(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, buffer.sampleRate, true);
  view.setUint32(28, buffer.sampleRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeStr(36, 'data');
  view.setUint32(40, numSamples * 2, true);

  for (let i = 0; i < numSamples; i++) {
    const s = Math.max(-1, Math.min(1, channelData[i]));
    view.setInt16(44 + i * 2, s < 0 ? s * 32768 : s * 32767, true);
  }

  return new Blob([out], { type: 'audio/wav' });
}

function parseElevenLabsErrorDetail(status: number, rawText: string): {
  friendlyMessage: string;
  isUnusualActivityBlock: boolean;
} {
  const lower = rawText.toLowerCase();
  const isUnusualActivityBlock =
    lower.includes('detected_unusual_activity') ||
    lower.includes('free tier access has been disabled') ||
    lower.includes('proxy or vpn');

  if (isUnusualActivityBlock) {
    return {
      isUnusualActivityBlock: true,
      friendlyMessage:
        'ElevenLabs 401 Block (detected_unusual_activity): Free Tier access is disabled by ElevenLabs when a VPN/Proxy or shared IP is detected. Disable VPN, use a paid ElevenLabs key, or use the Built-in Acoustic Synth / Gemini Audio Fallback.'
    };
  }

  return {
    isUnusualActivityBlock: false,
    friendlyMessage: `ElevenLabs API Error (${status}): ${rawText}`
  };
}

/**
 * Calls POST https://api.elevenlabs.io/v1/speech-to-text with Scribe v2.
 * Includes automatic audio extraction, Khmer language_code retry (khm -> auto-detect),
 * and model_id retry (scribe_v2 -> scribe_v1).
 */
export async function transcribeWithElevenLabsScribe2(
  mediaFile: File,
  apiKey: string,
  languageHint?: 'en' | 'km',
  onProgress?: (operation: string) => void
): Promise<ScribeTranscriptionResult> {
  if (!apiKey.trim()) {
    throw new Error('ElevenLabs API key is required to invoke POST https://api.elevenlabs.io/v1/speech-to-text');
  }

  const uploadReadyFile = await extractCompactAudioWavFromFile(mediaFile, onProgress);

  const attemptConfigs: Array<{ modelId: string; langCode: string | null; label: string }> =
    languageHint === 'km'
      ? [
          { modelId: 'scribe_v2', langCode: 'khm', label: 'Scribe v2 (Khmer ISO-639-3: khm)' },
          { modelId: 'scribe_v2', langCode: null, label: 'Scribe v2 (Auto-Language Identification for Khmer)' },
          { modelId: 'scribe_v1', langCode: 'khm', label: 'Scribe v1 Compatibility (khm)' },
          { modelId: 'scribe_v1', langCode: null, label: 'Scribe v1 Compatibility (Auto-Detect)' }
        ]
      : [
          { modelId: 'scribe_v2', langCode: 'eng', label: 'Scribe v2 (English: eng)' },
          { modelId: 'scribe_v2', langCode: null, label: 'Scribe v2 (Auto-Language Identification)' },
          { modelId: 'scribe_v1', langCode: null, label: 'Scribe v1 Compatibility' }
        ];

  let lastErrorText = '';
  let lastStatus = 500;

  for (const attempt of attemptConfigs) {
    onProgress?.(`Uploading audio & executing ${attempt.label}...`);

    const formData = new FormData();
    formData.append('file', uploadReadyFile);
    formData.append('model_id', attempt.modelId);
    formData.append('diarize', 'true');
    formData.append('tag_audio_events', 'true');
    formData.append('timestamps_granularity', 'word');
    if (attempt.langCode) {
      formData.append('language_code', attempt.langCode);
    }

    const response = await fetch('https://api.elevenlabs.io/v1/speech-to-text', {
      method: 'POST',
      headers: {
        'xi-api-key': apiKey.trim()
      },
      body: formData
    });

    if (response.ok) {
      const rawData = await response.json();
      const rawResponseJson = JSON.stringify(rawData, null, 2);
      onProgress?.('Normalizing Scribe word-level timestamps & Khmer/English script segmentation...');
      return normalizeScribePayloadToInternalSchema(rawData, rawResponseJson, languageHint);
    }

    lastStatus = response.status;
    lastErrorText = await response.text();
    const parsedErr = parseElevenLabsErrorDetail(lastStatus, lastErrorText);

    if (lastStatus === 401 || lastStatus === 403) {
      throw new ElevenLabsAccountRestrictionError(
        parsedErr.friendlyMessage,
        lastStatus,
        parsedErr.isUnusualActivityBlock
      );
    }
  }

  const finalParsed = parseElevenLabsErrorDetail(lastStatus, lastErrorText);
  throw new Error(finalParsed.friendlyMessage);
}

/**
 * Normalizes Scribe v2 response into WordEvents, DialogueSegments, and CharacterProfiles.
 * Handles both space-delimited English tokens and continuous Cambodian Khmer script (ក-៿).
 */
export function normalizeScribePayloadToInternalSchema(
  rawData: Record<string, unknown>,
  rawResponseJson: string,
  languageHint?: 'en' | 'km'
): ScribeTranscriptionResult {
  const fullText = String(rawData.text || '').trim();
  const hasKhmerScript = /[ក-៿]/.test(fullText);
  const langCodeRaw = String(rawData.language_code || '').toLowerCase();

  const detectedLanguage: 'en' | 'km' =
    hasKhmerScript ||
    langCodeRaw.startsWith('kh') ||
    langCodeRaw.startsWith('km') ||
    languageHint === 'km'
      ? 'km'
      : 'en';
  const targetLanguage: 'km' | 'en' = detectedLanguage === 'en' ? 'km' : 'en';

  const rawWords = Array.isArray(rawData.words) ? rawData.words : [];

  const segments: DialogueSegment[] = [];
  let currentWords: WordEvent[] = [];
  let currentAudioEvents: string[] = [];
  let currentSpeaker = 'speaker_0';
  let segCounter = 1;
  let wordCounter = 1;

  const joinWordEventsNaturally = (words: WordEvent[]): string => {
    if (words.length === 0) return '';
    if (detectedLanguage === 'km') {
      let out = '';
      words.forEach((w, idx) => {
        const token = w.original_text.trim();
        if (idx === 0) {
          out += token;
        } else {
          const prev = words[idx - 1];
          const pause = w.start - prev.end;
          if (pause > 0.15 || /[a-zA-Z0-9]$/.test(out) || /^[a-zA-Z0-9]/.test(token)) {
            out += ' ' + token;
          } else {
            out += token;
          }
        }
      });
      return out.trim();
    }

    return words
      .map((w) => w.original_text)
      .join(' ')
      .replace(/\s+([.,!?;:])/g, '$1')
      .trim();
  };

  const flushSegment = () => {
    if (currentWords.length === 0) return;
    const segId = `seg_${segCounter++}`;
    const start = Number(currentWords[0].start.toFixed(3));
    const end = Number(Math.max(start + 0.4, currentWords[currentWords.length - 1].end).toFixed(3));
    const availableDuration = Number(Math.max(0.4, end - start).toFixed(3));
    const text = joinWordEventsNaturally(currentWords);

    const charId = `char_${currentSpeaker}`;
    const updatedWords = currentWords.map((w) => ({ ...w, segment_id: segId }));

    segments.push({
      segment_id: segId,
      speaker_id: currentSpeaker,
      character_id: charId,
      source_start: start,
      source_end: end,
      available_duration: availableDuration,
      proposed_dubbed_start: start,
      proposed_dubbed_end: end,
      actual_synthesized_duration: null,
      final_approved_start: start,
      final_approved_end: end,
      playback_rate_adjustment: 1.0,
      original_text: text,
      language: detectedLanguage,
      target_language: targetLanguage,
      word_events: updatedWords,
      audio_events: [...currentAudioEvents],
      translated_text: '',
      literal_meaning: '',
      natural_spoken_version: '',
      shorter_dubbing_version: '',
      selected_version_type: 'natural',
      active_target_text: '',
      emotion_direction: '',
      delivery_notes: '',
      pronunciation_notes: '',
      estimated_speech_duration: null,
      source_engine: SourceEngine.SCRIBE_V2,
      review_status: ReviewStatus.VERIFIED,
      review_required: false,
      review_reason: '',
      timing_status: TimingStatus.UNMEASURED,
      overlap_seconds: 0,
      synthesized_audio_base64: null,
      synthesized_audio_mime: null,
      synthesized_voice_id: null,
      is_locked_approved: false
    });

    currentWords = [];
    currentAudioEvents = [];
  };

  for (let i = 0; i < rawWords.length; i++) {
    const rw = rawWords[i] as Record<string, unknown>;
    const wType = String(rw.type || 'word') as 'word' | 'spacing' | 'audio_event';
    const wText = String(rw.text || '').trim();
    if (!wText || wType === 'spacing') continue;

    const wStart = typeof rw.start === 'number' ? rw.start : 0;
    const wEnd = typeof rw.end === 'number' ? rw.end : wStart + 0.25;
    const speaker = String(rw.speaker_id ?? currentSpeaker);

    if (wType === 'audio_event') {
      currentAudioEvents.push(wText);
      continue;
    }

    const prevWord = currentWords[currentWords.length - 1];
    const pauseGap = prevWord ? wStart - prevWord.end : 0;
    const speakerChanged = currentWords.length > 0 && speaker !== currentSpeaker;
    const sentenceEnded =
      prevWord && /[.!?។៕]$/.test(prevWord.original_text) && currentWords.length >= 3;
    const naturalPauseExceeded = pauseGap > 0.95 && currentWords.length >= 2;
    const maxContextReached =
      currentWords.length >= (detectedLanguage === 'km' ? 28 : 22) ||
      (currentWords.length > 0 && wEnd - currentWords[0].start > 6.5);

    if (speakerChanged || sentenceEnded || naturalPauseExceeded || maxContextReached) {
      flushSegment();
    }

    currentSpeaker = speaker;
    const confidence =
      typeof rw.confidence === 'number'
        ? rw.confidence
        : typeof rw.logprob === 'number'
          ? Number(Math.min(0.99, Math.max(0.1, Math.exp(rw.logprob))).toFixed(2))
          : undefined;

    currentWords.push({
      word_id: `w_${wordCounter++}`,
      segment_id: '',
      speaker_id: speaker,
      start: wStart,
      end: wEnd,
      original_text: wText,
      language: detectedLanguage,
      word_type: 'word',
      confidence,
      source_engine: SourceEngine.SCRIBE_V2,
      review_status: confidence !== undefined && confidence < 0.72 ? ReviewStatus.PENDING : ReviewStatus.VERIFIED
    });
  }

  flushSegment();

  if (segments.length === 0 && fullText.length > 0) {
    const rawSentences = fullText
      .split(/(?<=[.!?។៕])\s+|\n+/)
      .map((s) => s.trim())
      .filter(Boolean);

    let cursorSec = 0.5;
    rawSentences.forEach((sentence, idx) => {
      const segId = `seg_${idx + 1}`;
      const estDur = Number(Math.max(1.8, Math.min(6.0, sentence.length * 0.085)).toFixed(2));
      const start = Number(cursorSec.toFixed(2));
      const end = Number((start + estDur).toFixed(2));
      cursorSec = end + 0.4;

      const wordObj: WordEvent = {
        word_id: `w_${idx + 1}_1`,
        segment_id: segId,
        speaker_id: 'speaker_0',
        start,
        end,
        original_text: sentence,
        language: detectedLanguage,
        word_type: 'word',
        source_engine: SourceEngine.SCRIBE_V2,
        review_status: ReviewStatus.PENDING
      };

      segments.push({
        segment_id: segId,
        speaker_id: 'speaker_0',
        character_id: 'char_speaker_0',
        source_start: start,
        source_end: end,
        available_duration: estDur,
        proposed_dubbed_start: start,
        proposed_dubbed_end: end,
        actual_synthesized_duration: null,
        final_approved_start: start,
        final_approved_end: end,
        playback_rate_adjustment: 1.0,
        original_text: sentence,
        language: detectedLanguage,
        target_language: targetLanguage,
        word_events: [wordObj],
        audio_events: [],
        translated_text: '',
        literal_meaning: '',
        natural_spoken_version: '',
        shorter_dubbing_version: '',
        selected_version_type: 'natural',
        active_target_text: '',
        emotion_direction: '',
        delivery_notes: '',
        pronunciation_notes: '',
        estimated_speech_duration: null,
        source_engine: SourceEngine.SCRIBE_V2,
        review_status: ReviewStatus.PENDING,
        review_required: true,
        review_reason: 'Constructed from Scribe top-level Khmer text; verify word timing.',
        timing_status: TimingStatus.UNMEASURED,
        overlap_seconds: 0,
        synthesized_audio_base64: null,
        synthesized_audio_mime: null,
        synthesized_voice_id: null,
        is_locked_approved: false
      });
    });
  }

  const speakerMap = new Map<string, CharacterProfile>();
  const defaultVoices = DEFAULT_FALLBACK_VOICES;

  segments.forEach((seg, idx) => {
    const existing = speakerMap.get(seg.speaker_id);
    if (!existing) {
      const voicePreset = defaultVoices[speakerMap.size % defaultVoices.length];
      speakerMap.set(seg.speaker_id, {
        character_id: seg.character_id,
        original_speaker_id: seg.speaker_id,
        character_name: `Speaker ${seg.speaker_id.replace('speaker_', '#')}`,
        first_appearance: seg.source_start,
        dialogue_count: 1,
        voice_description: `Detected speaker ${seg.speaker_id} from Scribe diarization`,
        perceived_vocal_presentation: 'Diarized Speaker Track',
        relationship_context: 'Scene participant',
        assigned_voice_id: voicePreset.voice_id,
        assigned_voice_name: voicePreset.name,
        is_voice_locked: true,
        target_language_compatibility: 'verified_multilingual_v2',
        pronunciation_notes: '',
        voice_consistency_notes: 'Locked via Voice Vault',
        audition_audio_base64: null,
        review_status: ReviewStatus.VERIFIED
      });
    } else {
      existing.dialogue_count += 1;
    }

    const hasLowConf = seg.word_events.some((w) => w.confidence !== undefined && w.confidence < 0.72);
    if (hasLowConf) {
      segments[idx].review_required = true;
      segments[idx].review_reason = 'Contains low-confidence word tokens from Scribe transcription';
    }
  });

  return {
    rawResponseJson,
    detectedLanguage,
    segments,
    characters: Array.from(speakerMap.values())
  };
}

export async function fetchElevenLabsVoices(apiKey: string): Promise<ElevenLabsVoice[]> {
  if (!apiKey.trim()) {
    return DEFAULT_FALLBACK_VOICES;
  }

  const response = await fetch('https://api.elevenlabs.io/v1/voices', {
    headers: {
      'xi-api-key': apiKey.trim()
    }
  });

  if (!response.ok) {
    throw new Error(`Failed to fetch ElevenLabs voices (${response.status})`);
  }

  const data = await response.json();
  if (!Array.isArray(data.voices)) {
    return DEFAULT_FALLBACK_VOICES;
  }

  return data.voices.map((v: Record<string, unknown>) => ({
    voice_id: String(v.voice_id),
    name: String(v.name),
    category: String(v.category || 'custom'),
    labels: (v.labels as Record<string, string>) || {},
    preview_url: v.preview_url ? String(v.preview_url) : undefined,
    description: v.description ? String(v.description) : undefined
  }));
}

export async function synthesizeSpeechAndMeasureDuration(params: {
  text: string;
  voiceId: string;
  apiKey: string;
  modelId: string;
  ttsEngineMode?: TtsEngineMode;
  stability?: number;
  similarityBoost?: number;
}): Promise<{
  audioBase64: string;
  mimeType: string;
  actualDurationSeconds: number;
  engineUsed: 'elevenlabs' | 'local_acoustic_fallback';
  fallbackReason?: string;
}> {
  const {
    text,
    voiceId,
    apiKey,
    modelId,
    ttsEngineMode = 'elevenlabs_with_fallback',
    stability = 0.55,
    similarityBoost = 0.78
  } = params;

  if (!text.trim()) {
    throw new Error('Cannot synthesize empty dialogue text.');
  }

  if (ttsEngineMode === 'local_acoustic_synth' || !apiKey.trim()) {
    const localWav = await generateAcousticRehearsalWav(text, voiceId);
    return {
      ...localWav,
      engineUsed: 'local_acoustic_fallback'
    };
  }

  const response = await fetch(
    `https://api.elevenlabs.io/v1/text-to-speech/${encodeURIComponent(voiceId)}?output_format=mp3_44100_128`,
    {
      method: 'POST',
      headers: {
        'xi-api-key': apiKey.trim(),
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        text: text.trim(),
        model_id: modelId || 'eleven_multilingual_v2',
        voice_settings: {
          stability,
          similarity_boost: similarityBoost,
          style: 0.15,
          use_speaker_boost: true
        }
      })
    }
  );

  if (!response.ok) {
    const errText = await response.text();
    const parsedErr = parseElevenLabsErrorDetail(response.status, errText);

    if (ttsEngineMode === 'elevenlabs_with_fallback') {
      const localWav = await generateAcousticRehearsalWav(text, voiceId);
      return {
        ...localWav,
        engineUsed: 'local_acoustic_fallback',
        fallbackReason: parsedErr.friendlyMessage
      };
    }

    throw new ElevenLabsAccountRestrictionError(
      parsedErr.friendlyMessage,
      response.status,
      parsedErr.isUnusualActivityBlock
    );
  }

  const arrayBuffer = await response.arrayBuffer();
  const actualDurationSeconds = await measureArrayBufferAudioDuration(arrayBuffer);
  const audioBase64 = arrayBufferToBase64(arrayBuffer);

  return {
    audioBase64,
    mimeType: 'audio/mpeg',
    actualDurationSeconds,
    engineUsed: 'elevenlabs'
  };
}

export async function measureArrayBufferAudioDuration(arrayBuffer: ArrayBuffer): Promise<number> {
  const audioCtx = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();
  try {
    const copy = arrayBuffer.slice(0);
    const audioBuffer = await audioCtx.decodeAudioData(copy);
    return Number(audioBuffer.duration.toFixed(3));
  } finally {
    await audioCtx.close();
  }
}

function arrayBufferToBase64(buffer: ArrayBuffer): string {
  const bytes = new Uint8Array(buffer);
  let binary = '';
  for (let i = 0; i < bytes.byteLength; i++) {
    binary += String.fromCharCode(bytes[i]);
  }
  return btoa(binary);
}

export async function generateAcousticRehearsalWav(
  text: string,
  voiceId: string
): Promise<{
  audioBase64: string;
  mimeType: string;
  actualDurationSeconds: number;
}> {
  const cleanChars = text.replace(/\s+/g, '').length;
  const durationSec = Number(Math.max(1.1, Math.min(9.5, cleanChars * 0.068)).toFixed(3));
  const sampleRate = 24000;
  const numSamples = Math.floor(sampleRate * durationSec);
  const buffer = new ArrayBuffer(44 + numSamples * 2);
  const view = new DataView(buffer);

  const writeStr = (offset: number, str: string) => {
    for (let i = 0; i < str.length; i++) {
      view.setUint8(offset + i, str.charCodeAt(i));
    }
  };

  writeStr(0, 'RIFF');
  view.setUint32(4, 36 + numSamples * 2, true);
  writeStr(8, 'WAVE');
  writeStr(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeStr(36, 'data');
  view.setUint32(40, numSamples * 2, true);

  let baseFreq = 140;
  for (let i = 0; i < voiceId.length; i++) {
    baseFreq += voiceId.charCodeAt(i) % 7;
  }

  for (let i = 0; i < numSamples; i++) {
    const t = i / sampleRate;
    const syllableEnvelope = Math.max(0, Math.sin(t * Math.PI * 5.2));
    const f0 = baseFreq + Math.sin(t * 6) * 18;
    const sample =
      (Math.sin(2 * Math.PI * f0 * t) * 0.6 +
        Math.sin(2 * Math.PI * f0 * 2 * t) * 0.25 +
        Math.sin(2 * Math.PI * f0 * 3 * t) * 0.15) *
      syllableEnvelope *
      0.32;
    const pcm = Math.max(-1, Math.min(1, sample)) * 32767;
    view.setInt16(44 + i * 2, pcm, true);
  }

  const actualDurationSeconds = await measureArrayBufferAudioDuration(buffer);
  return {
    audioBase64: arrayBufferToBase64(buffer),
    mimeType: 'audio/wav',
    actualDurationSeconds
  };
}
