  // Subtitle & Audio Monitor Verification State
  const [subtitleConfig, setSubtitleConfig] = useState<SubtitleDisplayConfig>({

    audioMonitorSource: 'dubbed_audio',
    burnInSubtitlesOnExport: true
  });

  // Language & Style Settings
  const [sourceLanguage, setSourceLanguage] = useState<'en' | 'km'>('en');
  const [targetLanguage, setTargetLanguage] = useState<'km' | 'en'>('km');

  // Core Pipeline Data + Undo/Redo History
  const [rawScribeResponseJson, setRawScribeResponseJson] = useState<string | null>(null);
  const [segments, setSegments] = useState<DialogueSegment[]>([]);
  const [undoStack, setUndoStack] = useState<DialogueSegment[][]>([]);
  const [redoStack, setRedoStack] = useState<DialogueSegment[][]>([]);

  const [segmentSynthesisErrors, setSegmentSynthesisErrors] = useState<Record<string, string>>({});

  const [isAutoPipelineRunning, setIsAutoPipelineRunning] = useState<boolean>(false);

  const [audioMixMode, setAudioMixMode] = useState<AudioMixMode>(AudioMixMode.DUCK_ORIGINAL);

  const [verificationTests, setVerificationTests] = useState<VerificationTestResult[]>([]);

  const [elevenLabsRestrictionNotice, setElevenLabsRestrictionNotice] = useState<string | null>(null);

  // Workspace Navigation & Modals

  const [showRawScribeModal, setShowRawScribeModal] = useState<boolean>(false);

  // Credentials
  const [credentials, setCredentials] = useState<ApiCredentials>(() => {
    const saved = localStorage.getItem('angkordub_credentials_v2');
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        return {
          customGeminiApiKey: parsed.customGeminiApiKey || '',
          elevenLabsApiKey: parsed.elevenLabsApiKey || '',
          whisperEndpointUrl: parsed.whisperEndpointUrl || 'https://api.openai.com/v1/audio/transcriptions',
          whisperApiKey: parsed.whisperApiKey || '',
          whisperModel: parsed.whisperModel || 'whisper-1',
          persistLocally: Boolean(parsed.persistLocally),
          elevenLabsTtsModel: parsed.elevenLabsTtsModel || 'eleven_multilingual_v2',
          ttsEngineMode: parsed.ttsEngineMode || 'elevenlabs_with_fallback',
          enableScribeFallbackToGemini: parsed.enableScribeFallbackToGemini ?? true,
          enableWhisperVerification: Boolean(parsed.enableWhisperVerification),
          lipSyncEndpointUrl: parsed.lipSyncEndpointUrl || ''
        };
      } catch {
        // ignore invalid JSON
      }
    }
    return {
      customGeminiApiKey: '',
      elevenLabsApiKey: '',
      whisperEndpointUrl: 'https://api.openai.com/v1/audio/transcriptions',
      whisperApiKey: '',
      whisperModel: 'whisper-1',
      persistLocally: false,
      elevenLabsTtsModel: 'eleven_multilingual_v2',
      ttsEngineMode: 'elevenlabs_with_fallback',
      enableScribeFallbackToGemini: true,
      enableWhisperVerification: false,
      lipSyncEndpointUrl: ''
    };
  });

  // Real Stage Progress Trackers
  const [stageProgress, setStageProgress] = useState<Record<string, StageProgress>>({
    scribe: { status: 'waiting', operation: 'Ready for media', processed: 0, total: 1, startedAt: null, completedAt: null },
    whisper: { status: 'waiting', operation: 'Optional verification', processed: 0, total: 1, startedAt: null, completedAt: null },
    gemini: { status: 'waiting', operation: 'Waiting for transcript', processed: 0, total: 1, startedAt: null, completedAt: null },
    synthesis: { status: 'waiting', operation: 'Waiting for localization', processed: 0, total: 1, startedAt: null, completedAt: null },
    assembly: { status: 'waiting', operation: 'Waiting for verified audio', processed: 0, total: 100, startedAt: null, completedAt: null }
  });

  const [logs, setLogs] = useState<ProductionLog[]>([]);

  const videoRef = useRef<HTMLVideoElement | null>(null);
  const audioPreviewRef = useRef<HTMLAudioElement | null>(null);
  const lastTriggeredDubSegmentRef = useRef<string | null>(null);
  const sliceStopTimeoutRef = useRef<number | null>(null);

  const appendLog = (
    stage: ProductionLog['stage'],
    level: ProductionLog['level'],
    message: string,
    details?: string
  ) => {
    setLogs((prev) => [
      {
        id: `log_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
        timestamp: new Date().toLocaleTimeString(),
        stage,
        level,
        message,
        details
      },
      ...prev.slice(0, 149)
    ]);
  };

  const commitSegmentsUpdate = (updater: (prev: DialogueSegment[]) => DialogueSegment[]) => {
    setSegments((prev) => {
      const next = updater(prev);
      setUndoStack((u) => [...u.slice(-19), prev]);
      setRedoStack([]);
      setHasUnsavedTimelineEdits(true);
      return next;
    });
  };

  const handleUndo = () => {
    if (undoStack.length === 0) return;
    const previous = undoStack[undoStack.length - 1];
    setRedoStack((r) => [segments, ...r]);
    setUndoStack((u) => u.slice(0, -1));
    setSegments(previous);
    appendLog('SYSTEM', 'info', 'Undid last edit.');
  };

  const handleRedo = () => {
    if (redoStack.length === 0) return;
    const next = redoStack[0];
    setUndoStack((u) => [...u, segments]);
    setRedoStack((r) => r.slice(1));
    setSegments(next);
    appendLog('SYSTEM', 'info', 'Redid edit.');
  };

  // Restore saved checkpoint from IndexedDB on mount
  useEffect(() => {
    let mounted = true;
    loadCheckpointFromIndexedDB().then(async (checkpoint) => {
      if (!mounted) return;
      if (checkpoint && checkpoint.segments.length > 0) {
        setProjectName(checkpoint.projectName);
        setSourceLanguage(checkpoint.sourceLanguage);
        setTargetLanguage(checkpoint.targetLanguage);
        setDialogueStyle(checkpoint.dialogueStyle);
        setLipSyncMode(checkpoint.lipSyncMode);
        setAudioMixMode(checkpoint.audioMixMode);
        setRawScribeResponseJson(checkpoint.rawScribeResponseJson);
        setSegments(checkpoint.segments);
        setCharacters(checkpoint.characters);
        setDisagreements(checkpoint.disagreements);
        setVideoDurationSec(checkpoint.sourceVideoDuration || 16);
        setSelectedSegmentId(checkpoint.segments[0]?.segment_id || null);
        setNeedsVideoReselect(true);

        const localizedCount = checkpoint.segments.filter((s) => Boolean(s.active_target_text)).length;
        const measuredCount = checkpoint.segments.filter(
          (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
        ).length;

        setStageProgress({
          scribe: {
            status: 'completed',
            operation: `Transcription: ${checkpoint.segments.length} of ${checkpoint.segments.length} segments complete.`,
            processed: checkpoint.segments.length,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: Date.now()
          },
          whisper: {
            status: 'ready',
            operation: 'Speaker verification ready',
            processed: checkpoint.characters.length,
            total: checkpoint.characters.length,
            startedAt: null,
            completedAt: null
          },
          gemini: {
            status: localizedCount === checkpoint.segments.length ? 'completed' : 'ready',
            operation: `Localization: ${localizedCount} of ${checkpoint.segments.length} segments complete.`,
            processed: localizedCount,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: null
          },
          synthesis: {
            status: measuredCount === checkpoint.segments.length ? 'completed' : measuredCount > 0 ? 'completed_with_warnings' : 'ready',
            operation: `Voice generation: ${measuredCount} of ${checkpoint.segments.length} audio clips generated.`,
            processed: measuredCount,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: null
          },
          assembly: {
            status: measuredCount > 0 ? 'ready' : 'waiting',
            operation: measuredCount > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips',
            processed: 0,
            total: 100,
            startedAt: null,
            completedAt: null
          }
        });

        appendLog(
          'SYSTEM',
          'info',
          `Restored project "${checkpoint.projectName}" (${checkpoint.segments.length} segments, ${measuredCount} audio clips) from IndexedDB.`
        );
      } else {
        await handleLoadVerifiedDemoReel();
      }
    });

    fetchElevenLabsVoices(credentials.elevenLabsApiKey)
      .then((v) => {
        if (mounted) setAvailableVoices(v);
      })
      .catch(() => {});

    return () => {
      mounted = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // Auto-save checkpoint to IndexedDB when segments or characters change
  useEffect(() => {
    if (segments.length === 0) return;
    saveCheckpointToIndexedDB({
      projectName,
      updatedAt: new Date().toISOString(),
      sourceFileName: sourceFile?.name || null,
      sourceVideoDuration: videoDurationSec,
      sourceLanguage,
      targetLanguage,
      dialogueStyle,
      lipSyncMode,
      audioMixMode,
      rawScribeResponseJson,
      segments,
      characters,
      disagreements,
      logs: logs.slice(0, 30)
    }).catch(() => {});
  }, [
    projectName,
    sourceFile,
    videoDurationSec,
    sourceLanguage,
    targetLanguage,
    dialogueStyle,
    lipSyncMode,
    audioMixMode,
    rawScribeResponseJson,
    segments,
    characters,
    disagreements,
    logs
  ]);

  // Synchronized Dubbed Audio Monitor during video preview when "dubbed_audio" monitor mode is active
  useEffect(() => {
    if (!videoRef.current) return;
    videoRef.current.muted = subtitleConfig.audioMonitorSource === 'dubbed_audio';
  }, [subtitleConfig.audioMonitorSource]);

  const handleVideoTimeUpdate = () => {
    if (!videoRef.current) return;
    const t = videoRef.current.currentTime;
    setCurrentTimeSec(t);

    if (subtitleConfig.audioMonitorSource === 'dubbed_audio' && isPlaying) {
      const matchingDubSeg = segments.find(
        (s) =>
          s.synthesized_audio_base64 &&
          !s.needs_resynthesis &&
          t >= s.final_approved_start &&
          t <= s.final_approved_end
      );
      if (
        matchingDubSeg &&
        lastTriggeredDubSegmentRef.current !== matchingDubSeg.segment_id
      ) {
        lastTriggeredDubSegmentRef.current = matchingDubSeg.segment_id;
        playBase64Audio(
          matchingDubSeg.synthesized_audio_base64!,
          matchingDubSeg.synthesized_audio_mime,
          matchingDubSeg.playback_rate_adjustment || 1.0
        );
      } else if (!matchingDubSeg) {
        lastTriggeredDubSegmentRef.current = null;
      }
    }
  };

  // Play exact original video/audio window for a segment (for A/B comparison)
  const handlePlayOriginalWindow = (seg: DialogueSegment) => {
    if (!videoRef.current) return;
    if (sliceStopTimeoutRef.current) {
      window.clearTimeout(sliceStopTimeoutRef.current);
    }
    videoRef.current.muted = false;
    videoRef.current.currentTime = seg.source_start;
    videoRef.current.play().catch(() => {});

    const durMs = Math.max(400, (seg.source_end - seg.source_start) * 1000);
    sliceStopTimeoutRef.current = window.setTimeout(() => {
      if (videoRef.current) {
        videoRef.current.pause();
        videoRef.current.muted = subtitleConfig.audioMonitorSource === 'dubbed_audio';
      }
    }, durMs);
  };

  const handleSaveCredentials = (updated: ApiCredentials) => {
    setCredentials(updated);
    setElevenLabsRestrictionNotice(null);
    if (updated.persistLocally) {
      localStorage.setItem('angkordub_credentials_v2', JSON.stringify(updated));
    } else {
      localStorage.removeItem('angkordub_credentials_v2');
    }
    fetchElevenLabsVoices(updated.elevenLabsApiKey)
      .then((voices) => {
        setAvailableVoices(voices);
        appendLog('VOICE_VAULT', 'success', `Synced ${voices.length} voices from ElevenLabs.`);
      })
      .catch((err) => {
        appendLog('VOICE_VAULT', 'warning', `Voice sync note: ${err.message}`);
      });
  };

  // Load Playable Verified Sample Movie with pre-generated Khmer translations & measurable audio
  const handleLoadVerifiedDemoReel = async () => {
    try {
      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'running',
          operation: 'Loading playable sample movie, Khmer translations & synthesized audio...',
          processed: 0,
          total: 4,
          startedAt: Date.now(),
          completedAt: null
        }
      }));
      const bundle = await generatePlayableDemoMovie((msg) => {
        appendLog('SCRIBE', 'info', msg);
      });

      setSourceFile(bundle.videoFile);
      setSourceVideoUrl(bundle.videoUrl);
      setNeedsVideoReselect(false);
      setVideoDurationSec(bundle.durationSec);
      setRawScribeResponseJson(JSON.stringify(bundle.rawScribePayload, null, 2));
      setSegments(bundle.segments);
      setCharacters(bundle.characters);
      setDisagreements(bundle.sampleDisagreements);
      setSelectedSegmentId(bundle.segments[0]?.segment_id || null);
      setFinalValidationReport(null);

      setStageProgress({
        scribe: {
          status: 'completed',
          operation: `Transcription: ${bundle.segments.length} of ${bundle.segments.length} segments complete.`,
          processed: bundle.segments.length,
          total: bundle.segments.length,
          startedAt: Date.now(),
          completedAt: Date.now()
        },
        whisper: {
          status: 'completed',
          operation: `${bundle.characters.length} speakers verified`,
          processed: bundle.characters.length,
          total: bundle.characters.length,
          startedAt: Date.now(),
          completedAt: Date.now()
        },
        gemini: {
          status: 'completed',
          operation: `Localization: ${bundle.segments.length} of ${bundle.segments.length} segments complete.`,
          processed: bundle.segments.length,
          total: bundle.segments.length,
          startedAt: Date.now(),
          completedAt: Date.now()
        },
        synthesis: {
          status: 'completed',
          operation: `Voice generation: ${bundle.segments.length} of ${bundle.segments.length} audio clips generated.`,
          processed: bundle.segments.length,
          total: bundle.segments.length,
          startedAt: Date.now(),
          completedAt: Date.now()
        },
        assembly: {
          status: 'ready',
          operation: 'Ready to assemble and export movie',
          processed: 0,
          total: 100,
          startedAt: null,
          completedAt: null
        }
      });

      appendLog(
        'SCRIBE',
        'success',
        `Loaded sample movie "${bundle.videoFile.name}" with ${bundle.segments.length} Khmer segments and ready-to-audition audio clips.`
      );
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      appendLog('SCRIBE', 'error', `Failed to generate demo movie: ${msg}`);
    }
  };

  // Import Custom Movie File
  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const url = URL.createObjectURL(file);
    setSourceFile(file);
    setSourceVideoUrl(url);
    setNeedsVideoReselect(false);
    setProjectName(file.name.replace(/\.[^/.]+$/, ''));
    setActiveStage('STAGE_1_TRANSCRIBE');
    setStageProgress((p) => ({
      ...p,
      scribe: {
        status: 'ready',
        operation: `Ready to transcribe "${file.name}"`,
        processed: 0,
        total: 1,
        startedAt: null,
        completedAt: null
      }
    }));
    appendLog('SYSTEM', 'info', `Imported movie "${file.name}" (${(file.size / 1024 / 1024).toFixed(2)} MB).`);
  };

  // Gemini Audio Transcription Fallback
  const handleRunGeminiAudioTranscription = async (): Promise<DialogueSegment[] | null> => {
    if (!sourceFile) {
      appendLog('SCRIBE', 'error', 'Please import a movie file first.');
      return null;
    }

    try {
      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'running',
          operation: `Transcribing ${sourceLanguage === 'km' ? 'Khmer' : 'English'} audio via Gemini 2.5 Flash...`,
          processed: 0,
          total: 1,
          startedAt: Date.now(),
          completedAt: null
        }
      }));

      const result = await transcribeAudioWithGeminiFallback(
        sourceFile,
        sourceLanguage,
        (op) => {
          setStageProgress((p) => ({
            ...p,
            scribe: { ...p.scribe, operation: op }
          }));
          appendLog('SCRIBE', 'info', op);
        },
        credentials.customGeminiApiKey
      );

      setRawScribeResponseJson(result.rawResponseJson);
      setSourceLanguage(result.detectedLanguage);
      setTargetLanguage(result.detectedLanguage === 'en' ? 'km' : 'en');
      setSegments(result.segments);
      setCharacters(result.characters);
      setSelectedSegmentId(result.segments[0]?.segment_id || null);

      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'completed',
          operation: `Transcription: ${result.segments.length} of ${result.segments.length} segments complete.`,
          processed: result.segments.length,
          total: result.segments.length,
          startedAt: p.scribe.startedAt,
          completedAt: Date.now()
        },
        gemini: {
          status: 'ready',
          operation: `Localization: 0 of ${result.segments.length} segments complete.`,
          processed: 0,
          total: result.segments.length,
          startedAt: null,
          completedAt: null
        }
      }));

      appendLog(
        'SCRIBE',
        'success',
        `Transcription complete: ${result.segments.length} segments and ${result.characters.length} speakers identified.`
      );
      return result.segments;
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setStageProgress((p) => ({
        ...p,
        scribe: {
          ...p.scribe,
          status: 'failed',
          operation: 'Transcription failed',
          errorMessage: msg
        }
      }));
      appendLog('SCRIBE', 'error', `Transcription failed: ${msg}`);
      return null;
    }
  };

  // Primary Scribe v2 Transcription with auto-fallback
  const handleRunLiveScribe2 = async (): Promise<DialogueSegment[] | null> => {
    if (!sourceFile) {
      appendLog('SCRIBE', 'error', 'Please import a movie file first.');
      return null;
    }

    if (!credentials.elevenLabsApiKey.trim()) {
      if (credentials.enableScribeFallbackToGemini) {
        return await handleRunGeminiAudioTranscription();
      }
      setIsConfigOpen(true);
      return null;
    }

    try {
      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'running',
          operation: `Transcribing with ElevenLabs Scribe v2...`,
          processed: 0,
          total: 1,
          startedAt: Date.now(),
          completedAt: null
        }
      }));

      const result = await transcribeWithElevenLabsScribe2(
        sourceFile,
        credentials.elevenLabsApiKey,
        sourceLanguage,
        (op) => {
          setStageProgress((p) => ({
            ...p,
            scribe: { ...p.scribe, operation: op }
          }));
          appendLog('SCRIBE', 'info', op);
        }
      );

      setRawScribeResponseJson(result.rawResponseJson);
      setSourceLanguage(result.detectedLanguage);
      setTargetLanguage(result.detectedLanguage === 'en' ? 'km' : 'en');
      setSegments(result.segments);
      setCharacters(result.characters);
      setSelectedSegmentId(result.segments[0]?.segment_id || null);

      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'completed',
          operation: `Transcription: ${result.segments.length} of ${result.segments.length} segments complete.`,
          processed: result.segments.length,
          total: result.segments.length,
          startedAt: p.scribe.startedAt,
          completedAt: Date.now()
        },
        gemini: {
          status: 'ready',
          operation: `Localization: 0 of ${result.segments.length} segments complete.`,
          processed: 0,
          total: result.segments.length,
          startedAt: null,
          completedAt: null
        }
      }));

      return result.segments;
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      if (err instanceof ElevenLabsAccountRestrictionError && err.isUnusualActivityBlock) {
        setElevenLabsRestrictionNotice(err.message);
      }

      if (credentials.enableScribeFallbackToGemini) {
        return await handleRunGeminiAudioTranscription();
      }

      setStageProgress((p) => ({
        ...p,
        scribe: {
          ...p.scribe,
          status: 'failed',
          operation: 'Transcription failed',
          errorMessage: msg
        }
      }));
      appendLog('SCRIBE', 'error', msg);
      return null;
    }
  };

        ...p,
        whisper: {
          status: 'running',
          operation: 'Running Whisper verification...',
          processed: 0,
          total: segments.length,
          startedAt: Date.now(),

        }

          apiKey: credentials.whisperApiKey,
          model: credentials.whisperModel
        },

      );

      setDisagreements(foundDisagreements);
      setStageProgress((p) => ({
        ...p,
        whisper: {
          status: 'completed',
          operation: `Verified (${foundDisagreements.length} differences)`,
          processed: segments.length,
          total: segments.length,
          startedAt: p.whisper.startedAt,
          completedAt: Date.now()
        }
      }));
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setStageProgress((p) => ({
        ...p,
        whisper: { ...p.whisper, status: 'failed', operation: 'Whisper error', errorMessage: msg }
      }));
      appendLog('WHISPER', 'error', msg);
    }
  };

  const handleResolveDisagreement = (
    disId: string,
    choice: 'scribe' | 'whisper' | 'custom',
    customText?: string
  ) => {
    const targetDis = disagreements.find((d) => d.id === disId);
    if (!targetDis) return;

    const finalText =
      choice === 'scribe'
        ? targetDis.scribe_text
        : choice === 'whisper'
          ? targetDis.whisper_text
          : customText || targetDis.proposed_correction;

    commitSegmentsUpdate((prev) =>
      prev.map((seg) =>
        seg.segment_id === targetDis.segment_id
          ? {
              ...seg,
              original_text: finalText,
              needs_retranslation: true,
              review_status: ReviewStatus.APPROVED
            }
          : seg
      )
    );

    setDisagreements((prev) =>
      prev.map((d) =>
        d.id === disId ? { ...d, resolved: true, resolution_choice: choice, proposed_correction: finalText } : d
      )
    );
  };

  // Stage 4: Single-Segment Synthesis with Immediate Audition Availability
  const handleSynthesizeSingleSegment = async (segmentId: string) => {
    const seg = segments.find((s) => s.segment_id === segmentId);
    if (!seg) return;

    const textToSpeak = seg.active_target_text || seg.natural_spoken_version || seg.original_text;
    const charProfile = characters.find((c) => c.character_id === seg.character_id);
    const voiceId = charProfile?.assigned_voice_id || 'pNInz6obpgDQGcFmaJgB';

    setSynthesizingSegmentId(segmentId);
    setSegmentSynthesisErrors((prev) => {
      const next = { ...prev };
      delete next[segmentId];
      return next;
    });

    try {
      const ttsResult = await synthesizeSpeechAndMeasureDuration({
        text: textToSpeak,
        voiceId,
        apiKey: credentials.elevenLabsApiKey,
        modelId: credentials.elevenLabsTtsModel,
        ttsEngineMode: credentials.ttsEngineMode
      });

      if (ttsResult.fallbackReason) {
        setElevenLabsRestrictionNotice(ttsResult.fallbackReason);
      }

      commitSegmentsUpdate((prev) => {
        const updated = prev.map((item) =>
          item.segment_id === segmentId
            ? {
                ...item,
                active_target_text: textToSpeak,
                synthesized_audio_base64: ttsResult.audioBase64,
                synthesized_audio_mime: ttsResult.mimeType,
                synthesized_voice_id: voiceId,
                synthesized_engine_used: ttsResult.engineUsed,
                actual_synthesized_duration: ttsResult.actualDurationSeconds,
                needs_resynthesis: false
              }
            : item
        );
        const report = evaluateTimelineDurationAlignment(updated);
        const readyClips = report.updatedSegments.filter(
          (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
        ).length;

        setStageProgress((p) => ({
          ...p,
          synthesis: {
            ...p.synthesis,
            status: readyClips === report.updatedSegments.length ? 'completed' : 'ready',
            processed: readyClips,
            total: report.updatedSegments.length,
            operation: `Voice generation: ${readyClips} of ${report.updatedSegments.length} audio clips generated.`
          },
          assembly: {
            ...p.assembly,
            status: readyClips > 0 ? 'ready' : 'waiting',
            operation: readyClips > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips'
          }
        }));

        return report.updatedSegments;
      });

      appendLog(
        'TTS_TIMING',
        'success',
        `Generated audio for ${seg.segment_id} (${ttsResult.actualDurationSeconds.toFixed(2)}s). Ready for immediate audition.`
      );
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setSegmentSynthesisErrors((prev) => ({ ...prev, [segmentId]: msg }));
      if (err instanceof ElevenLabsAccountRestrictionError && err.isUnusualActivityBlock) {
        setElevenLabsRestrictionNotice(err.message);
      }
      appendLog('TTS_TIMING', 'error', `Failed to synthesize ${segmentId}: ${msg}`);
    } finally {
      setSynthesizingSegmentId(null);
    }
  };

  // Batch Synthesis for All Unapproved or Outdated Segments
  const handleSynthesizeAllUnapprovedSegments = async (
    overrideSegments?: DialogueSegment[]
  ): Promise<DialogueSegment[] | null> => {
    const baseSegments = overrideSegments || segments;
    if (baseSegments.length === 0) return null;

    const toProcess = baseSegments.filter(
      (s) => !s.is_locked_approved && (!s.synthesized_audio_base64 || s.needs_resynthesis)
    );
    const targetBatch = toProcess.length > 0 ? toProcess : baseSegments.filter((s) => !s.is_locked_approved);

    if (targetBatch.length === 0) {
      appendLog('TTS_TIMING', 'info', 'All segments are already generated and approved.');
      return baseSegments;
    }

    setStageProgress((p) => ({
      ...p,
      synthesis: {
        status: 'running',
        operation: `Voice generation: 0 of ${targetBatch.length} audio clips generated.`,
        processed: 0,
        total: targetBatch.length,
        startedAt: Date.now(),
        completedAt: null
      }
    }));

    let workingSegments = [...baseSegments];
    let circuitBreakerTriggered = false;
    let activeMode = credentials.ttsEngineMode;

    for (let i = 0; i < targetBatch.length; i++) {
      if (cancelPipelineRef.current) break;

      const seg = targetBatch[i];
      setSynthesizingSegmentId(seg.segment_id);
      const textToSpeak = seg.active_target_text || seg.natural_spoken_version || seg.original_text;
      const charProfile = characters.find((c) => c.character_id === seg.character_id);
      const voiceId = charProfile?.assigned_voice_id || 'pNInz6obpgDQGcFmaJgB';

      setStageProgress((p) => ({
        ...p,
        synthesis: {
          ...p.synthesis,
          processed: i + 1,
          currentItemLabel: seg.segment_id,
          operation: `Voice generation: ${i + 1} of ${targetBatch.length} audio clips generated.`
        }
      }));

      try {
        const ttsResult = await synthesizeSpeechAndMeasureDuration({
          text: textToSpeak,
          voiceId,
          apiKey: credentials.elevenLabsApiKey,
          modelId: credentials.elevenLabsTtsModel,
          ttsEngineMode: activeMode
        });

        if (ttsResult.fallbackReason && !circuitBreakerTriggered) {
          circuitBreakerTriggered = true;
          activeMode = 'local_acoustic_synth';
          setElevenLabsRestrictionNotice(ttsResult.fallbackReason);
        }

        workingSegments = workingSegments.map((item) =>
          item.segment_id === seg.segment_id
            ? {
                ...item,
                active_target_text: textToSpeak,
                synthesized_audio_base64: ttsResult.audioBase64,
                synthesized_audio_mime: ttsResult.mimeType,
                synthesized_voice_id: voiceId,
                synthesized_engine_used: ttsResult.engineUsed,
                actual_synthesized_duration: ttsResult.actualDurationSeconds,
                needs_resynthesis: false
              }
            : item
        );
      } catch (err) {
        const msg = err instanceof Error ? err.message : String(err);
        setSegmentSynthesisErrors((prev) => ({ ...prev, [seg.segment_id]: msg }));
        appendLog('TTS_TIMING', 'error', `Error on ${seg.segment_id}: ${msg}`);
        if (err instanceof ElevenLabsAccountRestrictionError) {
          setElevenLabsRestrictionNotice(err.message);
          break;
        }
      }
    }

    setSynthesizingSegmentId(null);
    const timingReport = evaluateTimelineDurationAlignment(workingSegments);
    setSegments(timingReport.updatedSegments);

    const readyCount = timingReport.updatedSegments.filter(
      (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
    ).length;

    setStageProgress((p) => ({
      ...p,
      synthesis: {
        status: timingReport.failedOverflowCount > 0 ? 'completed_with_warnings' : 'completed',
        operation: `Voice generation: ${readyCount} of ${timingReport.updatedSegments.length} audio clips generated.`,
        processed: readyCount,
        total: timingReport.updatedSegments.length,
        startedAt: p.synthesis.startedAt,
        completedAt: Date.now()
      },
      assembly: {
        status: readyCount > 0 ? 'ready' : 'waiting',
        operation: readyCount > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips',
        processed: 0,
        total: 100,
        startedAt: null,
        completedAt: null
      }
    }));

    return timingReport.updatedSegments;
  };

  const handleAutoFitAllOverflows = () => {
    const overflowSegs = segments.filter((s) => s.timing_status === TimingStatus.OVERFLOW_FAILED);
    if (overflowSegs.length === 0) return;

    commitSegmentsUpdate((prev) => {
      const adjusted = prev.map((s) => {
        if (s.timing_status !== TimingStatus.OVERFLOW_FAILED) return s;
        return {
          ...s,
          selected_version_type: s.shorter_dubbing_version ? ('shorter' as const) : s.selected_version_type,
          active_target_text: s.shorter_dubbing_version || s.active_target_text,
          playback_rate_adjustment: Math.min(1.15, Math.max(1.1, s.playback_rate_adjustment))
        };
      });
      return evaluateTimelineDurationAlignment(adjusted).updatedSegments;
    });
    appendLog('TTS_TIMING', 'success', `Applied concise wording & 1.10x pacing to ${overflowSegs.length} overflowing segments.`);
  };

    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setStageProgress((p) => ({

        assembly: { ...p.assembly, status: 'failed', operation: msg, errorMessage: msg }
      }));
      appendLog('ASSEMBLY', 'error', msg);
    }
  };

  // Auto-Run End-to-End Pipeline Controller
  const handleRunAutomatedPipeline = async () => {
    if (isAutoPipelineRunning) {
      cancelPipelineRef.current = true;
      setIsAutoPipelineRunning(false);
      appendLog('SYSTEM', 'warning', 'Auto-Run paused by user.');
      return;
    }

    cancelPipelineRef.current = false;
    setIsAutoPipelineRunning(true);

    try {
      let currentSegs = segments;

      if (currentSegs.length === 0) {
        setActiveStage('STAGE_1_TRANSCRIBE');
        const transcribed = await handleRunLiveScribe2();
        if (!transcribed || cancelPipelineRef.current) {
          setIsAutoPipelineRunning(false);
          return;
        }
        currentSegs = transcribed;
      }

      const needsLoc = currentSegs.some((s) => !s.active_target_text || s.needs_retranslation);
      if (needsLoc && !cancelPipelineRef.current) {
        setActiveStage('STAGE_3_LOCALIZE_KHMER');
        const localized = await handleRunGeminiLocalization(currentSegs);
        if (!localized || cancelPipelineRef.current) {
          setIsAutoPipelineRunning(false);
          return;
        }
        currentSegs = localized;
      }

      const needsSynth = currentSegs.some(
        (s) => !s.synthesized_audio_base64 || s.needs_resynthesis
      );
      if (needsSynth && !cancelPipelineRef.current) {
        setActiveStage('STAGE_4_VOICES_AUDITION');
        const synthesized = await handleSynthesizeAllUnapprovedSegments(currentSegs);
        if (!synthesized || cancelPipelineRef.current) {
          setIsAutoPipelineRunning(false);
          return;
        }
        currentSegs = synthesized;
      }

      if (!cancelPipelineRef.current) {
        setActiveStage('STAGE_6_ASSEMBLE_EXPORT');
        await handleRunFinalAssembly(currentSegs);
      }
    } finally {
      setIsAutoPipelineRunning(false);
    }
  };

  // Timeline Interactive Handlers
  const handleUpdateSegmentTiming = (
    segmentId: string,
    newStart: number,
    newEnd: number,
    target: 'dubbed' | 'source'
  ) => {
    const clampedStart = Math.max(0, Math.min(videoDurationSec - 0.25, Number(newStart.toFixed(2))));
    const clampedEnd = Math.max(clampedStart + 0.25, Math.min(videoDurationSec, Number(newEnd.toFixed(2))));

    commitSegmentsUpdate((prev) => {
      const updated = prev.map((seg) => {
        if (seg.segment_id !== segmentId) return seg;
        if (target === 'source') {
          if (seg.source_timing_locked !== false) return seg;
          return {
            ...seg,
            source_start: clampedStart,
            source_end: clampedEnd,
            available_duration: Number((clampedEnd - clampedStart).toFixed(3))
          };
        }
        return {
          ...seg,
          proposed_dubbed_start: clampedStart,
          proposed_dubbed_end: clampedEnd,
          final_approved_start: clampedStart,
          final_approved_end: clampedEnd
        };
      });
      return evaluateTimelineDurationAlignment(updated).updatedSegments;
    });
  };

  const handleSplitSegmentAtPlayhead = (segmentId: string, splitTimeSec: number) => {
    const targetSeg = segments.find((s) => s.segment_id === segmentId);
    if (!targetSeg) return;

    const midTime =
      splitTimeSec > targetSeg.source_start + 0.3 && splitTimeSec < targetSeg.source_end - 0.3
        ? Number(splitTimeSec.toFixed(2))
        : Number(((targetSeg.source_start + targetSeg.source_end) / 2).toFixed(2));

    const wordsA = targetSeg.original_text.split(' ');
    const halfIdx = Math.max(1, Math.floor(wordsA.length / 2));
    const textPart1 = wordsA.slice(0, halfIdx).join(' ') || targetSeg.original_text;
    const textPart2 = wordsA.slice(halfIdx).join(' ') || '...';

    commitSegmentsUpdate((prev) => {
      const idx = prev.findIndex((s) => s.segment_id === segmentId);
      if (idx === -1) return prev;

      const firstHalf: DialogueSegment = {
        ...targetSeg,
        segment_id: `${targetSeg.segment_id}_a`,
        source_end: midTime,
        available_duration: Number((midTime - targetSeg.source_start).toFixed(2)),
        proposed_dubbed_end: midTime,
        final_approved_end: midTime,
        original_text: textPart1,
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const secondHalf: DialogueSegment = {
        ...targetSeg,
        segment_id: `${targetSeg.segment_id}_b`,
        source_start: midTime,
        available_duration: Number((targetSeg.source_end - midTime).toFixed(2)),
        proposed_dubbed_start: midTime,
        final_approved_start: midTime,
        original_text: textPart2,
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const next = [...prev.slice(0, idx), firstHalf, secondHalf, ...prev.slice(idx + 1)];
      return evaluateTimelineDurationAlignment(next).updatedSegments;
    });
  };

  const handleMergeSelectedSegments = () => {

    if (idx === -1 || idx >= segments.length - 1) return;

    const second = segments[idx + 1];

    commitSegmentsUpdate((prev) => {
      const merged: DialogueSegment = {
        ...first,

        proposed_dubbed_end: second.proposed_dubbed_end,
        final_approved_end: second.final_approved_end,
        original_text: `${first.original_text} ${second.original_text}`.trim(),
        active_target_text: `${first.active_target_text} ${second.active_target_text}`.trim(),
        natural_spoken_version: `${first.natural_spoken_version} ${second.natural_spoken_version}`.trim(),
        word_events: [...first.word_events, ...second.word_events],
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const next = [...prev.slice(0, idx), merged, ...prev.slice(idx + 2)];
      return evaluateTimelineDurationAlignment(next).updatedSegments;
    });
  };

  const handleAuditionCharacterVoice = async (charId: string) => {
    const char = characters.find((c) => c.character_id === charId);
    if (!char) return;

      targetLanguage === 'km'
        ? 'សូមប្រុងប្រយ័ត្នបន្តិចបង ផែនទីប្រាសាទបុរាណនេះសំខាន់ណាស់សម្រាប់ពួកយើង។'
        : 'Listen carefully my friend, the gates of Angkor only open before sunrise.';

    try {

      });

      if (res.fallbackReason) {
        setElevenLabsRestrictionNotice(res.fallbackReason);
      }

      setCharacters((prev) =>
        prev.map((c) =>
          c.character_id === charId ? { ...c, audition_audio_base64: res.audioBase64 } : c

      );

      playBase64Audio(res.audioBase64, res.mimeType);

    const raw = base64.trim();

      : `data:${mimeType || 'audio/wav'};base64,${raw}`;

    audioPreviewRef.current.play().catch(() => {});

          c.character_id === targetCharId

            : c
        );
    });

                : 'Waiting',
      summary: `${localizedSegsCount} of ${totalSegs} translated`

                : 'Waiting',

            ? 'Completed'

            Dismiss

                  <button

                                }}
                                className="px-3 py-1 rounded-lg bg-emerald-500/20 hover:bg-emerald-500/30 border border-emerald-500/40 text-xs font-semibold text-emerald-200 flex items-center gap-1"
                              >
                                <Volume2 className="w-3.5 h-3.5" /> Audition Khmer
                              </button>
                            )}
                          </div>
                        </div>

                        <textarea
                          rows={2}
                          value={seg.active_target_text}
                          onClick={(e) => e.stopPropagation()}
                          onChange={(e) =>
                            handleInlineEditSegmentText(
                              seg.segment_id,
                              'active_target_text',
                              e.target.value
                            )
                          }
                          placeholder="Khmer translation..."
                          className="w-full p-2.5 rounded-lg bg-[#0B0C10] border border-[#262B3D] text-base text-white font-['Kantumruy_Pro',sans-serif] leading-relaxed focus:outline-none focus:border-[#D4AF37]"
                        />
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}

          {/* STAGE 5: SYNCHRONIZE TIMING & SUBTITLES */}
          {activeStage === 'STAGE_5_SYNCHRONIZE' && (
            <div className="flex-1 flex flex-col overflow-hidden">
              <div className="px-6 py-4 border-b border-[#222738] bg-[#11141D] flex flex-wrap items-center justify-between gap-3">
                <div>
                  <h2 className="text-base font-bold text-white">
                    5. Synchronize Audio Timing & Verify Lip-Sync Alignment
                  </h2>
                  <p className="text-xs text-slate-400">
                    Resolve duration overflows before assembly and open the multitrack timeline for fine adjustments.
                  </p>
                </div>

                <div className="flex items-center gap-2">
                  {overflowSegsCount > 0 && (
                    <button
                      onClick={handleAutoFitAllOverflows}
                      className="px-3.5 py-1.5 rounded-lg bg-amber-500/20 hover:bg-amber-500/30 border border-amber-500/40 text-amber-200 text-xs font-semibold"
                    >
                      Auto-Fit {overflowSegsCount} Overflows (Concise + 1.10x)
                    </button>
                  )}
                  <button
                    onClick={() => setShowTimeline(!showTimeline)}
                    className="px-3.5 py-1.5 rounded-lg bg-[#181C28] hover:bg-[#222738] border border-[#2C3347] text-xs font-medium text-slate-200"
                  >
                    {showTimeline ? 'Hide Multitrack Timeline' : 'Open Multitrack Timeline'}
                  </button>
                  <button
                    onClick={() => setActiveStage('STAGE_6_ASSEMBLE_EXPORT')}
                    className="px-4 py-1.5 rounded-lg bg-[#D4AF37] text-[#0B0C10] text-xs font-bold"
                  >
                    Next: Assemble & Export →
                  </button>
                </div>
              </div>

              <div className="flex-1 overflow-y-auto p-6 space-y-3">
                {segments.map((seg) => {
                  const effectiveDur =
                    seg.actual_synthesized_duration !== null
                      ? Number((seg.actual_synthesized_duration / (seg.playback_rate_adjustment || 1.0)).toFixed(2))
                      : null;

                  return (
                    <div
                      key={seg.segment_id}
                      onClick={() => setSelectedSegmentId(seg.segment_id)}
                      className={`p-4 rounded-xl border flex flex-wrap items-center justify-between gap-4 cursor-pointer ${
                        seg.timing_status === TimingStatus.OVERFLOW_FAILED
                          ? 'bg-rose-950/15 border-rose-500/40'
                          : 'bg-[#11141D] border-[#222738]'
                      }`}
                    >
                      <div className="space-y-1 max-w-xl">
                        <div className="flex items-center gap-2 text-xs font-mono">
                          <span className="text-[#D4AF37] font-bold">{seg.segment_id}</span>
                          <span className="text-slate-400">
                            Source Window: {seg.available_duration.toFixed(2)}s
                          </span>
                          <span>→</span>
                          <span className="text-white">
                            Dubbed Duration: {effectiveDur !== null ? `${effectiveDur}s` : 'Unmeasured'}
                          </span>
                        </div>
                        <div className="text-sm text-white font-['Kantumruy_Pro',sans-serif]">
                          {seg.active_target_text || seg.original_text}
                        </div>
                      </div>

                      <div className="flex items-center gap-3" onClick={(e) => e.stopPropagation()}>
                        <div className="flex items-center gap-1.5 bg-[#0B0C10] px-2.5 py-1 rounded-lg border border-[#262B3D]">
                          <span className="text-xs text-slate-400">Speed:</span>
                          <select
                            value={seg.playback_rate_adjustment}
                            onChange={(e) => {
                              const rate = Number(e.target.value);
                              commitSegmentsUpdate((prev) => {
                                const updated = prev.map((s) =>
                                  s.segment_id === seg.segment_id
                                    ? { ...s, playback_rate_adjustment: rate }
                                    : s
                                );
                                return evaluateTimelineDurationAlignment(updated).updatedSegments;
                              });
                            }}
                            className="bg-transparent text-xs font-mono text-white focus:outline-none"
                          >
                            <option value={1.0}>1.00x (Natural)</option>
                            <option value={1.05}>1.05x</option>
                            <option value={1.1}>1.10x</option>
                            <option value={1.15}>1.15x</option>
                          </select>
                        </div>

                        <label className="flex items-center gap-1.5 text-xs text-slate-300 cursor-pointer">
                          <input
                            type="checkbox"
                            checked={Boolean(seg.lip_sync_manual_verified)}
                            onChange={(e) => {
                              const checked = e.target.checked;
                              commitSegmentsUpdate((prev) =>
                                prev.map((s) =>
                                  s.segment_id === seg.segment_id
                                    ? { ...s, lip_sync_manual_verified: checked }
                                    : s
                                )
                              );
                            }}
                            className="rounded border-slate-600 bg-[#0B0C10] text-emerald-400"
                          />
                          <span>Visual Sync Reviewed</span>
                        </label>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}

          {/* STAGE 6: ASSEMBLE & EXPORT WITH PRE-FLIGHT READINESS GATE */}
          {activeStage === 'STAGE_6_ASSEMBLE_EXPORT' && (
            <div className="flex-1 overflow-y-auto p-6 space-y-6">
              {/* Pre-Flight Readiness Check Banner (Prevents Assembly Blocker) */}
              <div
                className={`p-5 rounded-xl border ${
                  validAudioClipsCount > 0 && outdatedAudioClipsCount === 0
                    ? 'bg-emerald-500/10 border-emerald-500/30'
                    : 'bg-amber-500/10 border-amber-500/40'
                }`}
              >
                <div className="flex flex-wrap items-center justify-between gap-4">
                  <div className="space-y-1">
                    <div className="text-sm font-bold text-white flex items-center gap-2">
                      {validAudioClipsCount > 0 && outdatedAudioClipsCount === 0 ? (
                        <CheckCircle2 className="w-4 h-4 text-emerald-400" />
                      ) : (
                        <AlertTriangle className="w-4 h-4 text-amber-400" />
                      )}
                      <span>
                        Pre-Flight Assembly Readiness: {validAudioClipsCount} of {totalSegs} Valid Dubbed Audio Clips Ready
                      </span>
                    </div>
                    <p className="text-xs text-slate-300">
                      {validAudioClipsCount === 0
                        ? 'Final movie assembly requires synthesized dialogue audio. Click "Generate All Missing Audio Clips" below before assembling.'
                        : outdatedAudioClipsCount > 0
                          ? `${outdatedAudioClipsCount} segment(s) have edited Khmer text with outdated audio. Regenerate them so your export includes your latest edits.`
                          : 'All synthesized audio assets and timing anchors are verified and ready for final muxing.'}
                    </p>
                  </div>

                  {(validAudioClipsCount < totalSegs || outdatedAudioClipsCount > 0) && (
                    <button
                      onClick={() => handleSynthesizeAllUnapprovedSegments()}
                      disabled={stageProgress.synthesis.status === 'running'}
                      className="px-4 py-2 rounded-lg bg-[#D4AF37] text-[#0B0C10] text-xs font-bold flex items-center gap-1.5 shadow"
                    >
                      <Volume2 className="w-3.5 h-3.5" />
                      <span>
                        {stageProgress.synthesis.status === 'running'
                          ? 'Generating Audio Clips...'
                          : 'Generate Missing / Outdated Audio Clips Now'}
                      </span>
                    </button>
                  )}
                </div>
              </div>

              <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                {/* Export & Mixing Settings */}
                <div className="p-5 rounded-xl bg-[#11141D] border border-[#222738] space-y-4">
                  <h3 className="text-sm font-bold text-white">Audio Mixing & Subtitle Export Options</h3>

                  <div className="space-y-1.5">
                    <label className="block text-xs text-slate-300">Soundtrack Mixing Mode</label>
                    <select
                      value={audioMixMode}
                      onChange={(e) => setAudioMixMode(e.target.value as AudioMixMode)}
                      className="w-full px-3 py-2 rounded-lg bg-[#0B0C10] border border-[#2C3347] text-xs text-white"
                    >
                      <option value={AudioMixMode.DUCK_ORIGINAL}>
                        Auto-Duck Original Soundtrack Under Khmer Dialogue
                      </option>
                      <option value={AudioMixMode.REPLACE_ORIGINAL}>
                        Replace Original Audio Completely with Khmer Dialogue
                      </option>
                      <option value={AudioMixMode.DIALOGUE_ONLY}>
                        Export Dubbed Khmer Dialogue Stem Only (.WAV)
                      </option>
                    </select>
                  </div>

                  <label className="flex items-center gap-2.5 cursor-pointer text-xs text-slate-200">
                    <input
                      type="checkbox"
                      checked={subtitleConfig.burnInSubtitlesOnExport}
                      onChange={(e) =>
                        setSubtitleConfig({
                          ...subtitleConfig,
                          burnInSubtitlesOnExport: e.target.checked
                        })
                      }
                      className="rounded border-slate-600 bg-[#0B0C10] text-[#D4AF37]"
                    />
                    <span>Burn-in subtitles on exported video (plus downloadable .SRT and .VTT files)</span>
                  </label>

                  <div className="pt-3 border-t border-[#1E2333]">
                    <button
                      onClick={() => handleRunFinalAssembly()}
                      disabled={
                        stageProgress.assembly.status === 'running' || validAudioClipsCount === 0
                      }
                      className="w-full py-2.5 rounded-xl bg-[#D4AF37] hover:bg-[#e3be42] text-[#0B0C10] font-bold text-xs uppercase tracking-wider flex items-center justify-center gap-2 shadow-lg disabled:opacity-40"
                    >
                      <Film className="w-4 h-4" />
                      <span>
                        {stageProgress.assembly.status === 'running'
                          ? stageProgress.assembly.operation
                          : validAudioClipsCount === 0
                            ? 'Generate Audio Clips First to Enable Assembly'
                            : 'Assemble & Validate Final Dubbed Movie'}
                      </span>
                    </button>
                  </div>
                </div>

                {/* Lip-Sync Mode Disclosure */}
                <div className="p-5 rounded-xl bg-[#11141D] border border-[#222738] space-y-3">
                  <h3 className="text-sm font-bold text-white">Synchronization & Lip-Sync Mode</h3>
                  <div className="p-3.5 rounded-lg bg-[#0B0C10] border border-[#222738] space-y-1">
                    <div className="text-xs font-semibold text-emerald-300">
                      Mode A: Timing-Aligned Dubbing (Active)
                    </div>
                    <p className="text-xs text-slate-400 leading-relaxed">
                      Places duration-measured Khmer speech accurately within Scribe&apos;s speaking windows. Visual mouth movements in the original video are not modified unless an external Mode B lip-sync worker is connected in Settings.
                    </p>
                  </div>
                </div>
              </div>

              {/* Completed Export & Playback Card */}
              {finalValidationReport && (
                <div className="p-6 rounded-xl bg-[#111622] border border-emerald-500/40 space-y-4">
                  <div className="flex flex-wrap items-center justify-between gap-4">
                    <div>
                      <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-semibold bg-emerald-500/20 text-emerald-300 mb-1">
                        <CheckCircle2 className="w-3.5 h-3.5" /> Final Movie Validated
                      </span>
                      <h3 className="text-base font-bold text-white">
                        Dubbed Movie Master Ready for Download
                      </h3>
                    </div>

                    <div className="flex flex-wrap items-center gap-2">
                      {finalValidationReport.srtDownloadUrl && (
                        <a
                          href={finalValidationReport.srtDownloadUrl}
                          download={`${projectName.replace(/\s+/g, '_')}.srt`}
                          className="px-3 py-2 rounded-lg bg-[#1A1E2B] hover:bg-[#252B3F] border border-[#2E364F] text-xs font-semibold text-sky-300 flex items-center gap-1.5"
                        >
                          <Download className="w-3.5 h-3.5" /> .SRT
                        </a>
                      )}
                      {finalValidationReport.vttDownloadUrl && (
                        <a
                          href={finalValidationReport.vttDownloadUrl}
                          download={`${projectName.replace(/\s+/g, '_')}.vtt`}
                          className="px-3 py-2 rounded-lg bg-[#1A1E2B] hover:bg-[#252B3F] border border-[#2E364F] text-xs font-semibold text-purple-300 flex items-center gap-1.5"
                        >
                          <Download className="w-3.5 h-3.5" /> .VTT
                        </a>
                      )}
                      <a
                        href={finalValidationReport.downloadUrl}
                        download={
                          finalValidationReport.mimeType.includes('wav')
                            ? `${projectName.replace(/\s+/g, '_')}_khmer_dialogue.wav`
                            : `${projectName.replace(/\s+/g, '_')}_dubbed_master.webm`
                        }
                        className="px-5 py-2 rounded-lg bg-emerald-500 hover:bg-emerald-400 text-[#0B0C10] font-bold text-xs flex items-center gap-2 shadow-lg"
                      >
                        <Download className="w-4 h-4" />
                        Download Dubbed Movie ({(finalValidationReport.fileSizeBytes / 1024).toFixed(0)} KB)
                      </a>
                    </div>
                  </div>

                  {!finalValidationReport.mimeType.includes('wav') && (
                    <video
                      src={finalValidationReport.downloadUrl}
                      controls
                      className="w-full max-w-2xl rounded-xl border border-[#262B3D] bg-black"
                    />
                  )}
                </div>
              )}
            </div>
          )}

          {/* AUDIT & QA REPORT WORKSPACE */}
          {activeStage === 'STUDIO_AUDIT_QA' && (
            <div className="flex-1 overflow-y-auto p-6 space-y-6">
              <div className="p-5 rounded-xl bg-[#11141D] border border-[#222738] space-y-4">
                <div className="flex items-center justify-between">
                  <div>
                    <h2 className="text-base font-bold text-white">
                      Production Quality Assurance & Blocking Diagnostics Report
                    </h2>
                    <p className="text-xs text-slate-400">
                      Distinguishes blocking export errors from non-blocking review warnings.
                    </p>
                  </div>
                  <button
                    onClick={() => {
                      const res = runStudioVerificationSuite(segments, characters);
                      setVerificationTests(res);
                    }}
                    className="px-3.5 py-1.5 rounded-lg bg-[#D4AF37] text-[#0B0C10] text-xs font-bold"
                  >
                    Run Acceptance Suite
                  </button>
                </div>

                {/* QA Checklist Grid */}
                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3 text-xs">
                  <div className="p-3.5 rounded-lg bg-[#0B0C10] border border-[#1E2333]">
                    <div className="text-slate-400">Synthesized Audio Clips</div>
                    <div className="text-base font-bold text-white mt-1">
                      {validAudioClipsCount} / {totalSegs} Ready
                    </div>
                    {validAudioClipsCount === 0 && (
                      <button
                        onClick={() => setActiveStage('STAGE_4_VOICES_AUDITION')}
                        className="text-rose-400 hover:underline mt-1 block font-medium"
                      >
                        Blocking: Go to Stage 4 →
                      </button>
                    )}
                  </div>

                  <div className="p-3.5 rounded-lg bg-[#0B0C10] border border-[#1E2333]">
                    <div className="text-slate-400">Outdated Audio Clips</div>
                    <div className="text-base font-bold text-white mt-1">
                      {outdatedAudioClipsCount} Stale
                    </div>
                    {outdatedAudioClipsCount > 0 && (
                      <button
                        onClick={() => setActiveStage('STAGE_4_VOICES_AUDITION')}
                        className="text-amber-300 hover:underline mt-1 block font-medium"
                      >
                        Regenerate edited lines →
                      </button>
                    )}
                  </div>

                  <div className="p-3.5 rounded-lg bg-[#0B0C10] border border-[#1E2333]">
                    <div className="text-slate-400">Khmer Localization</div>
                    <div className="text-base font-bold text-white mt-1">
                      {localizedSegsCount} / {totalSegs} Translated
                    </div>
                  </div>

                  <div className="p-3.5 rounded-lg bg-[#0B0C10] border border-[#1E2333]">
                    <div className="text-slate-400">Timing Overflows</div>
                    <div className="text-base font-bold text-white mt-1">
                      {overflowSegsCount} Overflows
                    </div>
                    {overflowSegsCount > 0 && (
                      <button
                        onClick={() => setActiveStage('STAGE_5_SYNCHRONIZE')}
                        className="text-amber-300 hover:underline mt-1 block font-medium"
                      >
                        Review in Stage 5 →
                      </button>
                    )}
                  </div>
                </div>
              </div>

              {/* Automated Verification Tests */}
              <div className="space-y-2.5">
                {verificationTests.map((t) => (
                  <div
                    key={t.id}
                    className={`p-4 rounded-xl border flex items-center justify-between ${
                      t.passed
                        ? 'bg-[#111622] border-emerald-500/35'
                        : 'bg-rose-950/20 border-rose-500/40'
                    }`}
                  >
                    <div>
                      <div className="text-sm font-bold text-white">{t.name}</div>
                      <p className="text-xs text-slate-400 mt-0.5">{t.details}</p>
                    </div>
                    <span
                      className={`px-2.5 py-1 rounded text-xs font-mono font-bold ${
                        t.passed ? 'bg-emerald-500/20 text-emerald-300' : 'bg-rose-500/20 text-rose-300'
                      }`}
                    >
                      {t.passed ? 'PASSED' : 'FAILED'}
                    </span>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>

      {/* AREA D: COLLAPSIBLE MULTITRACK TIMELINE EDITOR */}
      {showTimeline && (
        <MultiTrackTimeline
          segments={segments}
          characters={characters}
          durationSec={videoDurationSec}
          currentTimeSec={currentTimeSec}
          selectedSegmentId={selectedSegment?.segment_id || null}
          selectedSegmentIds={selectedSegmentIds}
          canUndo={undoStack.length > 0}
          canRedo={redoStack.length > 0}
          hasUnsavedTimelineEdits={hasUnsavedTimelineEdits}
          onSelectSegment={(id, multi) => {
            setSelectedSegmentId(id);
            if (multi) {
              setSelectedSegmentIds((prev) =>
                prev.includes(id) ? prev.filter((item) => item !== id) : [...prev, id]
              );
            } else {
              setSelectedSegmentIds([id]);
            }
          }}
          onSeek={(time) => {
            setCurrentTimeSec(time);
            if (videoRef.current) {
              videoRef.current.currentTime = time;
            }
          }}
          onUpdateSegmentTiming={handleUpdateSegmentTiming}
          onSplitSegmentAtPlayhead={handleSplitSegmentAtPlayhead}
          onMergeSelectedSegments={handleMergeSelectedSegments}
          onReassignSpeaker={handleReassignSpeaker}
          onInlineEditSegmentText={handleInlineEditSegmentText}
          onToggleSourceLock={handleToggleSourceLock}
          onUndo={handleUndo}
          onRedo={handleRedo}
        />
      )}

      {/* COLLAPSIBLE DIAGNOSTICS DRAWER (COMPACT 1-LINE FOOTER BY DEFAULT) */}
      <footer className="bg-[#08090C] border-t border-[#1F2433] shrink-0">
        <div className="px-5 py-1.5 flex items-center justify-between text-[11px] font-mono bg-[#0E1017]">
          <div className="flex items-center gap-3 min-w-0">
            <span className="text-slate-400 flex items-center gap-1.5 shrink-0">
              <Terminal className="w-3.5 h-3.5 text-[#D4AF37]" />
              Diagnostics ({logs.length})
            </span>
            {logs[0] && (
              <span className="text-slate-300 truncate">
                {logs[0].message}
              </span>
            )}
          </div>
          <button
            onClick={() => setShowConsoleLogs(!showConsoleLogs)}
            className="text-slate-400 hover:text-white shrink-0 ml-4"
          >
            {showConsoleLogs ? 'Hide Logs' : 'Show Logs'}
          </button>
        </div>
        {showConsoleLogs && (
          <div className="h-28 overflow-y-auto px-5 py-2 space-y-1 font-mono text-[11px]">
            {logs.map((log) => (
              <div key={log.id} className="flex items-start gap-2">
                <span className="text-slate-500 shrink-0">[{log.timestamp}]</span>
                <span className="px-1.5 rounded bg-white/5 text-[#D4AF37] shrink-0">{log.stage}</span>
                <span
                  className={
                    log.level === 'error'
                      ? 'text-rose-400'
                      : log.level === 'warning'
                        ? 'text-amber-300'
                        : log.level === 'success'
                          ? 'text-emerald-300'
                          : 'text-slate-300'
                  }
                >
                  {log.message}
                </span>
              </div>
            ))}
          </div>
        )}
      </footer>

      {/* API Configuration Modal */}
      <ApiConfigModal
        isOpen={isConfigOpen}
        onClose={() => setIsConfigOpen(false)}
        credentials={credentials}
        onSaveCredentials={handleSaveCredentials}
      />

      {/* Raw Scribe v2 JSON Modal */}
      {showRawScribeModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-4">
          <div className="bg-[#12151E] border border-[#262B3D] rounded-xl max-w-3xl w-full max-h-[80vh] flex flex-col overflow-hidden">
            <div className="px-5 py-3 border-b border-[#262B3D] flex items-center justify-between">
              <span className="text-xs font-mono font-bold text-[#D4AF37]">
                PRESERVED TRANSCRIPTION RAW PAYLOAD
              </span>
              <button
                onClick={() => setShowRawScribeModal(false)}
                className="text-xs text-slate-400 hover:text-white"
              >
                Close
              </button>
            </div>
            <pre className="p-4 text-xs font-mono text-emerald-300 overflow-auto flex-1 bg-[#0B0C10]">
              {rawScribeResponseJson || 'No payload recorded yet.'}
            </pre>
          </div>
        </div>
      )}
    </div>
  );
}

export default App;
