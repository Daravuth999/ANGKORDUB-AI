type ProductionStageTab =
  | 'STAGE_1_TRANSCRIBE'
  | 'STAGE_2_SPEAKERS'
  | 'STAGE_3_LOCALIZE_KHMER'
  | 'STAGE_4_VOICES_AUDITION'
  | 'STAGE_5_SYNCHRONIZE'
  | 'STAGE_6_ASSEMBLE_EXPORT'
  | 'STUDIO_AUDIT_QA';

type ConciseStageStatus = 'Waiting' | 'In Progress' | 'Needs Review' | 'Completed' | 'Failed';

export function App() {
  // Project & Media State (Strictly real user-imported media)
  const [projectName, setProjectName] = useState<string>('Untitled Dubbing Project');
  const [sourceFile, setSourceFile] = useState<File | null>(null);
  const [sourceVideoUrl, setSourceVideoUrl] = useState<string | null>(null);
  const [needsVideoReselect, setNeedsVideoReselect] = useState<boolean>(false);
  const [videoDurationSec, setVideoDurationSec] = useState<number>(0);
  const [currentTimeSec, setCurrentTimeSec] = useState<number>(0);
  const [isPlaying, setIsPlaying] = useState<boolean>(false);
  const [expandVideoPreview, setExpandVideoPreview] = useState<boolean>(false);

  // Collapsible Lower Timeline & Diagnostics Drawer
  const [showTimeline, setShowTimeline] = useState<boolean>(false);
  const [showConsoleLogs, setShowConsoleLogs] = useState<boolean>(false);

  // Subtitle & Audio Monitor Verification State
  const [subtitleConfig, setSubtitleConfig] = useState<SubtitleDisplayConfig>({
    visible: true,
    mode: 'both',
    fontSize: 'md',
    position: 'bottom',
    backgroundOpacity: 0.85,
    showSpeakerLabels: true,
    showTimestamps: true,
    audioMonitorSource: 'dubbed_audio',
    burnInSubtitlesOnExport: true
  });

  // Language & Style Settings
  const [sourceLanguage, setSourceLanguage] = useState<'en' | 'km'>('en');
  const [targetLanguage, setTargetLanguage] = useState<'km' | 'en'>('km');
  const [dialogueStyle, setDialogueStyle] = useState<DialogueStyle>(DialogueStyle.EVERYDAY_KHMER);
  const [sceneContextNotes, setSceneContextNotes] = useState<string>('');

  // Core Pipeline Data + Undo/Redo History
  const [rawScribeResponseJson, setRawScribeResponseJson] = useState<string | null>(null);
  const [segments, setSegments] = useState<DialogueSegment[]>([]);
  const [undoStack, setUndoStack] = useState<DialogueSegment[][]>([]);
  const [redoStack, setRedoStack] = useState<DialogueSegment[][]>([]);
  const [hasUnsavedTimelineEdits, setHasUnsavedTimelineEdits] = useState<boolean>(false);

  const [selectedSegmentId, setSelectedSegmentId] = useState<string | null>(null);
  const [selectedSegmentIds, setSelectedSegmentIds] = useState<string[]>([]);
  const [characters, setCharacters] = useState<CharacterProfile[]>([]);
  const [availableVoices, setAvailableVoices] = useState<ElevenLabsVoice[]>([]);
  const [disagreements, setDisagreements] = useState<WhisperDisagreement[]>([]);

  // Per-Segment Synthesis & Translation Loading State
  const [synthesizingSegmentId, setSynthesizingSegmentId] = useState<string | null>(null);
  const [translatingSegmentId, setTranslatingSegmentId] = useState<string | null>(null);
  const [segmentSynthesisErrors, setSegmentSynthesisErrors] = useState<Record<string, string>>({});

  // Actionable API Error Banners
  const [localizationErrorBanner, setLocalizationErrorBanner] = useState<string | null>(null);
  const [elevenLabsErrorBanner, setElevenLabsErrorBanner] = useState<string | null>(null);

  // Automated Pipeline Orchestration State
  const [isAutoPipelineRunning, setIsAutoPipelineRunning] = useState<boolean>(false);
  const cancelPipelineRef = useRef<boolean>(false);

  // Lip-Sync & Final Assembly Settings
  const [lipSyncMode, setLipSyncMode] = useState<LipSyncMode>(LipSyncMode.TIMING_ALIGNED);
  const [audioMixMode, setAudioMixMode] = useState<AudioMixMode>(AudioMixMode.DUCK_ORIGINAL);
  const [originalTrackGain, setOriginalTrackGain] = useState<number>(0.25);
  const [acknowledgeOverflows, setAcknowledgeOverflows] = useState<boolean>(true);
  const [finalValidationReport, setFinalValidationReport] = useState<FinalValidationReport | null>(null);
  const [verificationTests, setVerificationTests] = useState<VerificationTestResult[]>([]);

  // Workspace Navigation & Modals
  const [activeStage, setActiveStage] = useState<ProductionStageTab>('STAGE_1_TRANSCRIBE');
  const [isConfigOpen, setIsConfigOpen] = useState<boolean>(false);
  const [showRawScribeModal, setShowRawScribeModal] = useState<boolean>(false);

  // Credentials
  const [credentials, setCredentials] = useState<ApiCredentials>(() => {
    const saved = localStorage.getItem('angkordub_credentials_v2');
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        return {
          customGeminiApiKey: parsed.customGeminiApiKey || '',
          elevenLabsApiKey: parsed.elevenLabsApiKey || '',
          whisperEndpointUrl: parsed.whisperEndpointUrl || 'https://api.openai.com/v1/audio/transcriptions',
          whisperApiKey: parsed.whisperApiKey || '',
          whisperModel: parsed.whisperModel || 'whisper-1',
          persistLocally: Boolean(parsed.persistLocally),
          elevenLabsTtsModel: parsed.elevenLabsTtsModel || 'eleven_multilingual_v2',
          enableWhisperVerification: Boolean(parsed.enableWhisperVerification),
          lipSyncEndpointUrl: parsed.lipSyncEndpointUrl || ''
        };
      } catch {
        // ignore invalid JSON
      }
    }
    return {
      customGeminiApiKey: '',
      elevenLabsApiKey: '',
      whisperEndpointUrl: 'https://api.openai.com/v1/audio/transcriptions',
      whisperApiKey: '',
      whisperModel: 'whisper-1',
      persistLocally: false,
      elevenLabsTtsModel: 'eleven_multilingual_v2',
      enableWhisperVerification: false,
      lipSyncEndpointUrl: ''
    };
  });

  // Real Stage Progress Trackers
  const [stageProgress, setStageProgress] = useState<Record<string, StageProgress>>({
    scribe: { status: 'waiting', operation: 'Import a movie file to begin', processed: 0, total: 0, startedAt: null, completedAt: null },
    whisper: { status: 'waiting', operation: 'Optional verification', processed: 0, total: 0, startedAt: null, completedAt: null },
    gemini: { status: 'waiting', operation: 'Waiting for transcript', processed: 0, total: 0, startedAt: null, completedAt: null },
    synthesis: { status: 'waiting', operation: 'Waiting for localization', processed: 0, total: 0, startedAt: null, completedAt: null },
    assembly: { status: 'waiting', operation: 'Waiting for verified audio', processed: 0, total: 100, startedAt: null, completedAt: null }
  });

  const [logs, setLogs] = useState<ProductionLog[]>([]);

  const videoRef = useRef<HTMLVideoElement | null>(null);
  const audioPreviewRef = useRef<HTMLAudioElement | null>(null);
  const lastTriggeredDubSegmentRef = useRef<string | null>(null);
  const sliceStopTimeoutRef = useRef<number | null>(null);

  const appendLog = (
    stage: ProductionLog['stage'],
    level: ProductionLog['level'],
    message: string,
    details?: string
  ) => {
    setLogs((prev) => [
      {
        id: `log_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
        timestamp: new Date().toLocaleTimeString(),
        stage,
        level,
        message,
        details
      },
      ...prev.slice(0, 149)
    ]);
  };

  const commitSegmentsUpdate = (updater: (prev: DialogueSegment[]) => DialogueSegment[]) => {
    setSegments((prev) => {
      const next = updater(prev);
      setUndoStack((u) => [...u.slice(-19), prev]);
      setRedoStack([]);
      setHasUnsavedTimelineEdits(true);
      return next;
    });
  };

  const handleUndo = () => {
    if (undoStack.length === 0) return;
    const previous = undoStack[undoStack.length - 1];
    setRedoStack((r) => [segments, ...r]);
    setUndoStack((u) => u.slice(0, -1));
    setSegments(previous);
    appendLog('SYSTEM', 'info', 'Undid last edit.');
  };

  const handleRedo = () => {
    if (redoStack.length === 0) return;
    const next = redoStack[0];
    setUndoStack((u) => [...u, segments]);
    setRedoStack((r) => r.slice(1));
    setSegments(next);
    appendLog('SYSTEM', 'info', 'Redid edit.');
  };

  // Restore saved real project checkpoint from IndexedDB on mount
  useEffect(() => {
    let mounted = true;
    loadCheckpointFromIndexedDB().then(async (checkpoint) => {
      if (!mounted) return;
      if (checkpoint && checkpoint.segments.length > 0) {
        if (checkpoint.sourceFileName === 'angkor_temple_expedition_reel.webm') {
          await clearCheckpointInIndexedDB();
          return;
        }

        setProjectName(checkpoint.projectName);
        setSourceLanguage(checkpoint.sourceLanguage);
        setTargetLanguage(checkpoint.targetLanguage);
        setDialogueStyle(checkpoint.dialogueStyle);
        setLipSyncMode(checkpoint.lipSyncMode);
        setAudioMixMode(checkpoint.audioMixMode);
        setRawScribeResponseJson(checkpoint.rawScribeResponseJson);
        setSegments(checkpoint.segments);
        setCharacters(checkpoint.characters);
        setDisagreements(checkpoint.disagreements);
        setVideoDurationSec(checkpoint.sourceVideoDuration || 0);
        setSelectedSegmentId(checkpoint.segments[0]?.segment_id || null);
        setNeedsVideoReselect(true);

        const localizedCount = checkpoint.segments.filter((s) => Boolean(s.active_target_text)).length;
        const measuredCount = checkpoint.segments.filter(
          (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
        ).length;

        setStageProgress({
          scribe: {
            status: 'completed',
            operation: `Transcription: ${checkpoint.segments.length} of ${checkpoint.segments.length} segments complete.`,
            processed: checkpoint.segments.length,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: Date.now()
          },
          whisper: {
            status: 'ready',
            operation: 'Speaker verification ready',
            processed: checkpoint.characters.length,
            total: checkpoint.characters.length,
            startedAt: null,
            completedAt: null
          },
          gemini: {
            status: localizedCount === checkpoint.segments.length ? 'completed' : 'ready',
            operation: `Localization: ${localizedCount} of ${checkpoint.segments.length} segments complete.`,
            processed: localizedCount,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: null
          },
          synthesis: {
            status:
              measuredCount === checkpoint.segments.length
                ? 'completed'
                : measuredCount > 0
                  ? 'completed_with_warnings'
                  : 'waiting',
            operation: `Voice generation: ${measuredCount} of ${checkpoint.segments.length} audio clips generated.`,
            processed: measuredCount,
            total: checkpoint.segments.length,
            startedAt: null,
            completedAt: null
          },
          assembly: {
            status: measuredCount > 0 ? 'ready' : 'waiting',
            operation: measuredCount > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips',
            processed: 0,
            total: 100,
            startedAt: null,
            completedAt: null
          }
        });

        appendLog(
          'SYSTEM',
          'info',
          `Restored project "${checkpoint.projectName}" (${checkpoint.segments.length} segments, ${localizedCount} localized, ${measuredCount} audio clips).`
        );
      }
    });

    if (credentials.elevenLabsApiKey.trim()) {
      fetchElevenLabsVoices(credentials.elevenLabsApiKey)
        .then((v) => {
          if (mounted) setAvailableVoices(v);
        })
        .catch(() => {});
    }

    return () => {
      mounted = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // Auto-save checkpoint to IndexedDB when segments or characters change
  useEffect(() => {
    if (segments.length === 0) return;
    saveCheckpointToIndexedDB({
      projectName,
      updatedAt: new Date().toISOString(),
      sourceFileName: sourceFile?.name || null,
      sourceVideoDuration: videoDurationSec,
      sourceLanguage,
      targetLanguage,
      dialogueStyle,
      lipSyncMode,
      audioMixMode,
      rawScribeResponseJson,
      segments,
      characters,
      disagreements,
      logs: logs.slice(0, 30)
    }).catch(() => {});
  }, [
    projectName,
    sourceFile,
    videoDurationSec,
    sourceLanguage,
    targetLanguage,
    dialogueStyle,
    lipSyncMode,
    audioMixMode,
    rawScribeResponseJson,
    segments,
    characters,
    disagreements,
    logs
  ]);

  useEffect(() => {
    if (!videoRef.current) return;
    videoRef.current.muted = subtitleConfig.audioMonitorSource === 'dubbed_audio';
  }, [subtitleConfig.audioMonitorSource]);

  const handleVideoTimeUpdate = () => {
    if (!videoRef.current) return;
    const t = videoRef.current.currentTime;
    setCurrentTimeSec(t);

    if (subtitleConfig.audioMonitorSource === 'dubbed_audio' && isPlaying) {
      const matchingDubSeg = segments.find(
        (s) =>
          s.synthesized_audio_base64 &&
          !s.needs_resynthesis &&
          t >= s.final_approved_start &&
          t <= s.final_approved_end
      );
      if (
        matchingDubSeg &&
        lastTriggeredDubSegmentRef.current !== matchingDubSeg.segment_id
      ) {
        lastTriggeredDubSegmentRef.current = matchingDubSeg.segment_id;
        playBase64Audio(
          matchingDubSeg.synthesized_audio_base64!,
          matchingDubSeg.synthesized_audio_mime,
          matchingDubSeg.playback_rate_adjustment || 1.0
        );
      } else if (!matchingDubSeg) {
        lastTriggeredDubSegmentRef.current = null;
      }
    }
  };

  const handlePlayOriginalWindow = (seg: DialogueSegment) => {
    if (!videoRef.current) return;
    if (sliceStopTimeoutRef.current) {
      window.clearTimeout(sliceStopTimeoutRef.current);
    }
    videoRef.current.muted = false;
    videoRef.current.currentTime = seg.source_start;
    videoRef.current.play().catch(() => {});

    const durMs = Math.max(400, (seg.source_end - seg.source_start) * 1000);
    sliceStopTimeoutRef.current = window.setTimeout(() => {
      if (videoRef.current) {
        videoRef.current.pause();
        videoRef.current.muted = subtitleConfig.audioMonitorSource === 'dubbed_audio';
      }
    }, durMs);
  };

  const handleSaveCredentials = (updated: ApiCredentials) => {
    setCredentials(updated);
    setLocalizationErrorBanner(null);
    setElevenLabsErrorBanner(null);
    if (updated.persistLocally) {
      localStorage.setItem('angkordub_credentials_v2', JSON.stringify(updated));
    } else {
      localStorage.removeItem('angkordub_credentials_v2');
    }
    if (updated.elevenLabsApiKey.trim()) {
      fetchElevenLabsVoices(updated.elevenLabsApiKey)
        .then((voices) => {
          setAvailableVoices(voices);
          if (voices.length > 0) {
            setCharacters((prev) =>
              prev.map((c, idx) =>
                !c.assigned_voice_id
                  ? {
                      ...c,
                      assigned_voice_id: voices[idx % voices.length].voice_id,
                      assigned_voice_name: voices[idx % voices.length].name
                    }
                  : c
              )
            );
          }
          appendLog('VOICE_VAULT', 'success', `Synced ${voices.length} voices from ElevenLabs.`);
        })
        .catch((err) => {
          appendLog('VOICE_VAULT', 'error', `ElevenLabs voice sync failed: ${err.message}`);
        });
    }

    // If Stage 3 was previously failed or waiting on untranslated segments, automatically resume localization
    const hasUntranslated = segments.some((s) => !s.active_target_text.trim());
    if (segments.length > 0 && (stageProgress.gemini.status === 'failed' || hasUntranslated)) {
      handleRunGeminiLocalization(segments, updated.customGeminiApiKey);
    }
  };

  // Import Custom Movie File
  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const url = URL.createObjectURL(file);
    setSourceFile(file);
    setSourceVideoUrl(url);
    setNeedsVideoReselect(false);
    setProjectName(file.name.replace(/\.[^/.]+$/, ''));
    if (segments.length === 0) {
      setActiveStage('STAGE_1_TRANSCRIBE');
      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'ready',
          operation: `Ready to transcribe "${file.name}"`,
          processed: 0,
          total: 1,
          startedAt: null,
          completedAt: null
        }
      }));
    }
    appendLog('SYSTEM', 'info', `Loaded movie "${file.name}" (${(file.size / 1024 / 1024).toFixed(2)} MB).`);
  };

  // Primary Scribe v2 Transcription
  const handleRunLiveScribe2 = async (): Promise<DialogueSegment[] | null> => {
    if (!sourceFile) {
      appendLog('SCRIBE', 'error', 'Please import a movie file first.');
      return null;
    }

    if (!credentials.elevenLabsApiKey.trim()) {
      setIsConfigOpen(true);
      appendLog('SCRIBE', 'error', 'ElevenLabs API key is required for Scribe v2 transcription.');
      return null;
    }

    setElevenLabsErrorBanner(null);

    try {
      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'running',
          operation: 'Transcribing with ElevenLabs Scribe v2...',
          processed: 0,
          total: 1,
          startedAt: Date.now(),
          completedAt: null
        }
      }));

      const result = await transcribeWithElevenLabsScribe2(
        sourceFile,
        credentials.elevenLabsApiKey,
        sourceLanguage,
        (op) => {
          setStageProgress((p) => ({
            ...p,
            scribe: { ...p.scribe, operation: op }
          }));
          appendLog('SCRIBE', 'info', op);
        }
      );

      let voicesToUse = availableVoices;
      if (voicesToUse.length === 0 && credentials.elevenLabsApiKey.trim()) {
        try {
          voicesToUse = await fetchElevenLabsVoices(credentials.elevenLabsApiKey);
          setAvailableVoices(voicesToUse);
        } catch {
          // keep empty if fetch fails
        }
      }

      const assignedChars = result.characters.map((c, idx) => {
        if (voicesToUse.length > 0 && !c.assigned_voice_id) {
          const v = voicesToUse[idx % voicesToUse.length];
          return {
            ...c,
            assigned_voice_id: v.voice_id,
            assigned_voice_name: v.name,
            is_voice_locked: true
          };
        }
        return c;
      });

      setRawScribeResponseJson(result.rawResponseJson);
      setSourceLanguage(result.detectedLanguage);
      setTargetLanguage(result.detectedLanguage === 'en' ? 'km' : 'en');
      setSegments(result.segments);
      setCharacters(assignedChars);
      setSelectedSegmentId(result.segments[0]?.segment_id || null);

      setStageProgress((p) => ({
        ...p,
        scribe: {
          status: 'completed',
          operation: `Transcription: ${result.segments.length} of ${result.segments.length} segments complete.`,
          processed: result.segments.length,
          total: result.segments.length,
          startedAt: p.scribe.startedAt,
          completedAt: Date.now()
        },
        gemini: {
          status: 'ready',
          operation: `Localization: 0 of ${result.segments.length} segments complete.`,
          processed: 0,
          total: result.segments.length,
          startedAt: null,
          completedAt: null
        }
      }));

      appendLog(
        'SCRIBE',
        'success',
        `Scribe v2 completed: ${result.segments.length} segments and ${assignedChars.length} speakers extracted.`
      );
      return result.segments;
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      if (err instanceof ElevenLabsAccountRestrictionError) {
        setElevenLabsErrorBanner(err.message);
      }
      setStageProgress((p) => ({
        ...p,
        scribe: {
          ...p.scribe,
          status: 'failed',
          operation: 'Transcription failed',
          errorMessage: msg
        }
      }));
      appendLog('SCRIBE', 'error', msg);
      return null;
    }
  };

  // Optional Whisper Verification
  const handleRunWhisperVerification = async () => {
    if (segments.length === 0) return;

    if (!credentials.whisperApiKey.trim()) {
      setIsConfigOpen(true);
      appendLog('WHISPER', 'warning', 'Enter your Whisper API key in Settings to run second-pass verification.');
      return;
    }

    try {
      setStageProgress((p) => ({
        ...p,
        whisper: {
          status: 'running',
          operation: 'Running Whisper verification...',
          processed: 0,
          total: segments.length,
          startedAt: Date.now(),
          completedAt: null
        }
      }));

      const foundDisagreements = await runOptionalWhisperVerification(
        sourceFile,
        segments,
        {
          endpointUrl: credentials.whisperEndpointUrl,
          apiKey: credentials.whisperApiKey,
          model: credentials.whisperModel
        },
        (msg) => appendLog('WHISPER', 'info', msg)
      );

      setDisagreements(foundDisagreements);
      setStageProgress((p) => ({
        ...p,
        whisper: {
          status: 'completed',
          operation: `Verified (${foundDisagreements.length} differences)`,
          processed: segments.length,
          total: segments.length,
          startedAt: p.whisper.startedAt,
          completedAt: Date.now()
        }
      }));
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setStageProgress((p) => ({
        ...p,
        whisper: { ...p.whisper, status: 'failed', operation: 'Whisper error', errorMessage: msg }
      }));
      appendLog('WHISPER', 'error', msg);
    }
  };

  const handleResolveDisagreement = (
    disId: string,
    choice: 'scribe' | 'whisper' | 'custom',
    customText?: string
  ) => {
    const targetDis = disagreements.find((d) => d.id === disId);
    if (!targetDis) return;

    const finalText =
      choice === 'scribe'
        ? targetDis.scribe_text
        : choice === 'whisper'
          ? targetDis.whisper_text
          : customText || targetDis.proposed_correction;

    commitSegmentsUpdate((prev) =>
      prev.map((seg) =>
        seg.segment_id === targetDis.segment_id
          ? {
              ...seg,
              original_text: finalText,
              needs_retranslation: true,
              review_status: ReviewStatus.APPROVED
            }
          : seg
      )
    );

    setDisagreements((prev) =>
      prev.map((d) =>
        d.id === disId ? { ...d, resolved: true, resolution_choice: choice, proposed_correction: finalText } : d
      )
    );
  };

  // Single-Segment Khmer Localization
  const handleLocalizeSingleSegment = async (segmentId: string) => {
    const targetSeg = segments.find((s) => s.segment_id === segmentId);
    if (!targetSeg) return;

    setTranslatingSegmentId(segmentId);
    setLocalizationErrorBanner(null);

    try {
      const resList = await localizeDialogueSegmentsWithGemini({
        segments: [targetSeg],
        characters,
        dialogueStyle,
        sceneContextNotes,
        customGeminiApiKey: credentials.customGeminiApiKey
      });

      const loc = resList[0];
      if (loc) {
        commitSegmentsUpdate((prev) =>
          prev.map((seg) =>
            seg.segment_id === segmentId
              ? {
                  ...seg,
                  translated_text: loc.translated_text,
                  literal_meaning: loc.literal_meaning,
                  natural_spoken_version: loc.natural_spoken_version,
                  shorter_dubbing_version: loc.shorter_dubbing_version,
                  selected_version_type: 'natural',
                  active_target_text: loc.natural_spoken_version,
                  emotion_direction: loc.emotion_direction,
                  delivery_notes: loc.delivery_notes,
                  estimated_speech_duration: loc.estimated_speech_duration,
                  review_required: loc.review_required,
                  review_reason: loc.review_reason,
                  needs_retranslation: false,
                  needs_resynthesis: true
                }
              : seg
          )
        );
        appendLog('GEMINI', 'success', `Localized ${segmentId} into natural spoken Khmer.`);
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setLocalizationErrorBanner(msg);
      appendLog('GEMINI', 'error', `Failed to localize ${segmentId}: ${msg}`);
    } finally {
      setTranslatingSegmentId(null);
    }
  };

  // Stage 3: Full Cambodian Khmer Localization with Progressive Batch Saving
  const handleRunGeminiLocalization = async (
    overrideSegments?: DialogueSegment[],
    overrideGeminiKey?: string
  ): Promise<DialogueSegment[] | null> => {
    const workingInput = overrideSegments || segments;
    if (workingInput.length === 0) return null;

    setLocalizationErrorBanner(null);

    try {
      setStageProgress((p) => ({
        ...p,
        gemini: {
          status: 'running',
          operation: `Localization: 0 of ${workingInput.length} segments complete.`,
          processed: 0,
          total: workingInput.length,
          startedAt: Date.now(),
          completedAt: null
        }
      }));

      let progressiveSegments = [...workingInput];

      const localizedList = await localizeDialogueSegmentsWithGemini({
        segments: workingInput,
        characters,
        dialogueStyle,
        sceneContextNotes,
        customGeminiApiKey:
          overrideGeminiKey !== undefined ? overrideGeminiKey : credentials.customGeminiApiKey,
        onSegmentProgress: (done, total, segId) => {
          setStageProgress((p) => ({
            ...p,
            gemini: {
              ...p.gemini,
              processed: done,
              total,
              currentItemLabel: segId,
              operation: `Localization: ${done} of ${total} segments complete.`
            }
          }));
        },
        onBatchLocalized: (batchResults) => {
          const batchMap = new Map(batchResults.map((item) => [item.segment_id, item]));
          progressiveSegments = progressiveSegments.map((seg) => {
            const loc = batchMap.get(seg.segment_id);
            if (!loc) return seg;
            const textChanged = seg.active_target_text !== loc.natural_spoken_version;
            return {
              ...seg,
              translated_text: loc.translated_text,
              literal_meaning: loc.literal_meaning,
              natural_spoken_version: loc.natural_spoken_version,
              shorter_dubbing_version: loc.shorter_dubbing_version,
              selected_version_type: 'natural' as const,
              active_target_text: loc.natural_spoken_version,
              emotion_direction: loc.emotion_direction,
              delivery_notes: loc.delivery_notes,
              estimated_speech_duration: loc.estimated_speech_duration,
              review_required: loc.review_required,
              review_reason: loc.review_reason,
              needs_retranslation: false,
              needs_resynthesis: textChanged || !seg.synthesized_audio_base64
            };
          });
          setSegments(progressiveSegments);
        }
      });

      setStageProgress((p) => ({
        ...p,
        gemini: {
          status: 'completed',
          operation: `Localization: ${progressiveSegments.length} of ${progressiveSegments.length} segments complete.`,
          processed: progressiveSegments.length,
          total: progressiveSegments.length,
          startedAt: p.gemini.startedAt,
          completedAt: Date.now()
        }
      }));

      appendLog('GEMINI', 'success', `Localized ${localizedList.length} segments into natural spoken Khmer.`);
      return progressiveSegments;
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setLocalizationErrorBanner(msg);
      if (err instanceof GeminiBillingOrAuthError && err.requiresCustomKey) {
        setIsConfigOpen(true);
      }
      setStageProgress((p) => ({
        ...p,
        gemini: { ...p.gemini, status: 'failed', operation: 'Localization failed', errorMessage: msg }
      }));
      appendLog('GEMINI', 'error', `Localization error: ${msg}`);
      return null;
    }
  };

  // Stage 4: Single-Segment Synthesis with Immediate Audition Availability
  const handleSynthesizeSingleSegment = async (segmentId: string) => {
    const seg = segments.find((s) => s.segment_id === segmentId);
    if (!seg) return;

    const textToSpeak = (seg.active_target_text || seg.natural_spoken_version).trim();
    if (!textToSpeak) {
      const errMsg = `Segment ${segmentId} has not been localized to Khmer yet. Click "Translate Line" or "Translate All with Gemini" first.`;
      setSegmentSynthesisErrors((prev) => ({ ...prev, [segmentId]: errMsg }));
      appendLog('TTS_TIMING', 'error', errMsg);
      return;
    }

    const charProfile = characters.find((c) => c.character_id === seg.character_id);
    const voiceId = charProfile?.assigned_voice_id || availableVoices[0]?.voice_id || '';

    setSynthesizingSegmentId(segmentId);
    setElevenLabsErrorBanner(null);
    setSegmentSynthesisErrors((prev) => {
      const next = { ...prev };
      delete next[segmentId];
      return next;
    });

    try {
      const ttsResult = await synthesizeSpeechAndMeasureDuration({
        text: textToSpeak,
        voiceId,
        apiKey: credentials.elevenLabsApiKey,
        modelId: credentials.elevenLabsTtsModel
      });

      commitSegmentsUpdate((prev) => {
        const updated = prev.map((item) =>
          item.segment_id === segmentId
            ? {
                ...item,
                active_target_text: textToSpeak,
                synthesized_audio_base64: ttsResult.audioBase64,
                synthesized_audio_mime: ttsResult.mimeType,
                synthesized_voice_id: voiceId,
                synthesized_engine_used: ttsResult.engineUsed,
                actual_synthesized_duration: ttsResult.actualDurationSeconds,
                needs_resynthesis: false
              }
            : item
        );
        const report = evaluateTimelineDurationAlignment(updated);
        const readyClips = report.updatedSegments.filter(
          (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
        ).length;

        setStageProgress((p) => ({
          ...p,
          synthesis: {
            ...p.synthesis,
            status: readyClips === report.updatedSegments.length ? 'completed' : 'ready',
            processed: readyClips,
            total: report.updatedSegments.length,
            operation: `Voice generation: ${readyClips} of ${report.updatedSegments.length} audio clips generated.`
          },
          assembly: {
            ...p.assembly,
            status: readyClips > 0 ? 'ready' : 'waiting',
            operation: readyClips > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips'
          }
        }));

        return report.updatedSegments;
      });

      appendLog(
        'TTS_TIMING',
        'success',
        `Generated ElevenLabs audio for ${seg.segment_id} (${ttsResult.actualDurationSeconds.toFixed(2)}s). Ready for immediate audition.`
      );
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setSegmentSynthesisErrors((prev) => ({ ...prev, [segmentId]: msg }));
      if (err instanceof ElevenLabsAccountRestrictionError) {
        setElevenLabsErrorBanner(err.message);
      }
      appendLog('TTS_TIMING', 'error', `Failed to synthesize ${segmentId}: ${msg}`);
    } finally {
      setSynthesizingSegmentId(null);
    }
  };

  // Batch Synthesis for All Localized Unapproved Segments
  const handleSynthesizeAllUnapprovedSegments = async (
    overrideSegments?: DialogueSegment[]
  ): Promise<DialogueSegment[] | null> => {
    const baseSegments = overrideSegments || segments;
    if (baseSegments.length === 0) return null;

    if (!credentials.elevenLabsApiKey.trim()) {
      setIsConfigOpen(true);
      appendLog('TTS_TIMING', 'error', 'ElevenLabs API key is required to generate voice audio.');
      return null;
    }

    const localizedCandidates = baseSegments.filter((s) =>
      Boolean((s.active_target_text || s.natural_spoken_version).trim())
    );

    if (localizedCandidates.length === 0) {
      const errMsg = 'Cannot generate voice audio before Khmer localization completes. Please translate the dialogue in Stage 3 first.';
      appendLog('TTS_TIMING', 'error', errMsg);
      setStageProgress((p) => ({
        ...p,
        synthesis: {
          ...p.synthesis,
          status: 'waiting',
          operation: 'Waiting for Khmer localization'
        }
      }));
      return null;
    }

    const toProcess = localizedCandidates.filter(
      (s) => !s.is_locked_approved && (!s.synthesized_audio_base64 || s.needs_resynthesis)
    );
    const targetBatch = toProcess.length > 0 ? toProcess : localizedCandidates.filter((s) => !s.is_locked_approved);

    if (targetBatch.length === 0) {
      appendLog('TTS_TIMING', 'info', 'All localized segments are already generated and approved.');
      return baseSegments;
    }

    setElevenLabsErrorBanner(null);
    setStageProgress((p) => ({
      ...p,
      synthesis: {
        status: 'running',
        operation: `Voice generation: 0 of ${targetBatch.length} audio clips generated.`,
        processed: 0,
        total: targetBatch.length,
        startedAt: Date.now(),
        completedAt: null
      }
    }));

    let workingSegments = [...baseSegments];

    for (let i = 0; i < targetBatch.length; i++) {
      if (cancelPipelineRef.current) break;

      const seg = targetBatch[i];
      setSynthesizingSegmentId(seg.segment_id);
      const textToSpeak = (seg.active_target_text || seg.natural_spoken_version).trim();
      const charProfile = characters.find((c) => c.character_id === seg.character_id);
      const voiceId = charProfile?.assigned_voice_id || availableVoices[0]?.voice_id || '';

      setStageProgress((p) => ({
        ...p,
        synthesis: {
          ...p.synthesis,
          processed: i + 1,
          currentItemLabel: seg.segment_id,
          operation: `Voice generation: ${i + 1} of ${targetBatch.length} audio clips generated.`
        }
      }));

      try {
        const ttsResult = await synthesizeSpeechAndMeasureDuration({
          text: textToSpeak,
          voiceId,
          apiKey: credentials.elevenLabsApiKey,
          modelId: credentials.elevenLabsTtsModel
        });

        workingSegments = workingSegments.map((item) =>
          item.segment_id === seg.segment_id
            ? {
                ...item,
                active_target_text: textToSpeak,
                synthesized_audio_base64: ttsResult.audioBase64,
                synthesized_audio_mime: ttsResult.mimeType,
                synthesized_voice_id: voiceId,
                synthesized_engine_used: ttsResult.engineUsed,
                actual_synthesized_duration: ttsResult.actualDurationSeconds,
                needs_resynthesis: false
              }
            : item
        );
      } catch (err) {
        const msg = err instanceof Error ? err.message : String(err);
        setSegmentSynthesisErrors((prev) => ({ ...prev, [seg.segment_id]: msg }));
        appendLog('TTS_TIMING', 'error', `Error on ${seg.segment_id}: ${msg}`);
        if (err instanceof ElevenLabsAccountRestrictionError) {
          setElevenLabsErrorBanner(err.message);
          break;
        }
      }
    }

    setSynthesizingSegmentId(null);
    const timingReport = evaluateTimelineDurationAlignment(workingSegments);
    setSegments(timingReport.updatedSegments);

    const readyCount = timingReport.updatedSegments.filter(
      (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
    ).length;

    setStageProgress((p) => ({
      ...p,
      synthesis: {
        status:
          readyCount === 0
            ? 'failed'
            : timingReport.failedOverflowCount > 0
              ? 'completed_with_warnings'
              : 'completed',
        operation: `Voice generation: ${readyCount} of ${timingReport.updatedSegments.length} audio clips generated.`,
        processed: readyCount,
        total: timingReport.updatedSegments.length,
        startedAt: p.synthesis.startedAt,
        completedAt: Date.now()
      },
      assembly: {
        status: readyCount > 0 ? 'ready' : 'waiting',
        operation: readyCount > 0 ? 'Ready for movie assembly' : 'Waiting for required audio clips',
        processed: 0,
        total: 100,
        startedAt: null,
        completedAt: null
      }
    }));

    return readyCount > 0 ? timingReport.updatedSegments : null;
  };

  const handleAutoFitAllOverflows = () => {
    const overflowSegs = segments.filter((s) => s.timing_status === TimingStatus.OVERFLOW_FAILED);
    if (overflowSegs.length === 0) return;

    commitSegmentsUpdate((prev) => {
      const adjusted = prev.map((s) => {
        if (s.timing_status !== TimingStatus.OVERFLOW_FAILED) return s;
        return {
          ...s,
          selected_version_type: s.shorter_dubbing_version ? ('shorter' as const) : s.selected_version_type,
          active_target_text: s.shorter_dubbing_version || s.active_target_text,
          playback_rate_adjustment: Math.min(1.15, Math.max(1.1, s.playback_rate_adjustment))
        };
      });
      return evaluateTimelineDurationAlignment(adjusted).updatedSegments;
    });
    appendLog('TTS_TIMING', 'success', `Applied concise wording & 1.10x pacing to ${overflowSegs.length} overflowing segments.`);
  };

    const activeSegs = overrideSegments || segments;
    if (!sourceVideoUrl) {
      appendLog('ASSEMBLY', 'error', 'Source video stream is required for final assembly.');
      return;

    try {
      setStageProgress((p) => ({
        ...p,
        assembly: {
          status: 'running',
          operation: 'Assembling movie & mixing audio tracks...',
          processed: 5,
          total: 100,
          startedAt: Date.now(),
          completedAt: null

        audioMixMode,

        burnInSubtitles: subtitleConfig.burnInSubtitlesOnExport,

        onProgress: (pct, msg) => {
          setStageProgress((p) => ({
            ...p,
            assembly: {
              ...p.assembly,
              processed: pct,

            }

            : 'Completed with validation warnings',

        }
      }));

      setActiveStage('STAGE_6_ASSEMBLE_EXPORT');

        report.passedValidation ? 'success' : 'warning',
        `Movie assembly complete: ${report.resolution}, ${report.actualDurationSec}s.`
      );
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      setStageProgress((p) => ({
        ...p,
        assembly: { ...p.assembly, status: 'failed', operation: msg, errorMessage: msg }
      }));
      appendLog('ASSEMBLY', 'error', msg);
    }
  };

  // Auto-Run End-to-End Pipeline Controller
  const handleRunAutomatedPipeline = async () => {
    if (isAutoPipelineRunning) {
      cancelPipelineRef.current = true;
      setIsAutoPipelineRunning(false);
      appendLog('SYSTEM', 'warning', 'Auto-Run paused by user.');
      return;
    }

    if (!sourceFile && segments.length === 0) {
      appendLog('SYSTEM', 'error', 'Please import a movie file before starting Auto-Run.');
      return;
    }

    cancelPipelineRef.current = false;
    setIsAutoPipelineRunning(true);

    try {
      let currentSegs = segments;

      if (currentSegs.length === 0) {
        setActiveStage('STAGE_1_TRANSCRIBE');
        const transcribed = await handleRunLiveScribe2();
        if (!transcribed || cancelPipelineRef.current) {
          return;
        }
        currentSegs = transcribed;
      }

      const needsLoc = currentSegs.some((s) => !s.active_target_text || s.needs_retranslation);
      if (needsLoc && !cancelPipelineRef.current) {
        setActiveStage('STAGE_3_LOCALIZE_KHMER');
        const localized = await handleRunGeminiLocalization(currentSegs);
        if (!localized || cancelPipelineRef.current) {
          return;
        }
        currentSegs = localized;
      }

      const needsSynth = currentSegs.some(
        (s) => !s.synthesized_audio_base64 || s.needs_resynthesis
      );
      if (needsSynth && !cancelPipelineRef.current) {
        setActiveStage('STAGE_4_VOICES_AUDITION');
        const synthesized = await handleSynthesizeAllUnapprovedSegments(currentSegs);
        if (!synthesized || cancelPipelineRef.current) {
          return;
        }
        currentSegs = synthesized;
      }

      if (!cancelPipelineRef.current) {
        setActiveStage('STAGE_6_ASSEMBLE_EXPORT');
        await handleRunFinalAssembly(currentSegs);
      }
    } finally {
      setIsAutoPipelineRunning(false);
    }
  };

  // Timeline Interactive Handlers
  const handleUpdateSegmentTiming = (
    segmentId: string,
    newStart: number,
    newEnd: number,
    target: 'dubbed' | 'source'
  ) => {
    const safeMaxDur = Math.max(videoDurationSec, 1);
    const clampedStart = Math.max(0, Math.min(safeMaxDur - 0.25, Number(newStart.toFixed(2))));
    const clampedEnd = Math.max(clampedStart + 0.25, Math.min(safeMaxDur, Number(newEnd.toFixed(2))));

    commitSegmentsUpdate((prev) => {
      const updated = prev.map((seg) => {
        if (seg.segment_id !== segmentId) return seg;
        if (target === 'source') {
          if (seg.source_timing_locked !== false) return seg;
          return {
            ...seg,
            source_start: clampedStart,
            source_end: clampedEnd,
            available_duration: Number((clampedEnd - clampedStart).toFixed(3))
          };
        }
        return {
          ...seg,
          proposed_dubbed_start: clampedStart,
          proposed_dubbed_end: clampedEnd,
          final_approved_start: clampedStart,
          final_approved_end: clampedEnd
        };
      });
      return evaluateTimelineDurationAlignment(updated).updatedSegments;
    });
  };

  const handleSplitSegmentAtPlayhead = (segmentId: string, splitTimeSec: number) => {
    const targetSeg = segments.find((s) => s.segment_id === segmentId);
    if (!targetSeg) return;

    const midTime =
      splitTimeSec > targetSeg.source_start + 0.3 && splitTimeSec < targetSeg.source_end - 0.3
        ? Number(splitTimeSec.toFixed(2))
        : Number(((targetSeg.source_start + targetSeg.source_end) / 2).toFixed(2));

    const wordsA = targetSeg.original_text.split(' ');
    const halfIdx = Math.max(1, Math.floor(wordsA.length / 2));
    const textPart1 = wordsA.slice(0, halfIdx).join(' ') || targetSeg.original_text;
    const textPart2 = wordsA.slice(halfIdx).join(' ') || '...';

    commitSegmentsUpdate((prev) => {
      const idx = prev.findIndex((s) => s.segment_id === segmentId);
      if (idx === -1) return prev;

      const firstHalf: DialogueSegment = {
        ...targetSeg,
        segment_id: `${targetSeg.segment_id}_a`,
        source_end: midTime,
        available_duration: Number((midTime - targetSeg.source_start).toFixed(2)),
        proposed_dubbed_end: midTime,
        final_approved_end: midTime,
        original_text: textPart1,
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const secondHalf: DialogueSegment = {
        ...targetSeg,
        segment_id: `${targetSeg.segment_id}_b`,
        source_start: midTime,
        available_duration: Number((targetSeg.source_end - midTime).toFixed(2)),
        proposed_dubbed_start: midTime,
        final_approved_start: midTime,
        original_text: textPart2,
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const next = [...prev.slice(0, idx), firstHalf, secondHalf, ...prev.slice(idx + 1)];
      return evaluateTimelineDurationAlignment(next).updatedSegments;
    });
  };

  const handleMergeSelectedSegments = () => {
    if (!selectedSegmentId) return;
    const idx = segments.findIndex((s) => s.segment_id === selectedSegmentId);
    if (idx === -1 || idx >= segments.length - 1) return;

    const first = segments[idx];
    const second = segments[idx + 1];

    commitSegmentsUpdate((prev) => {
      const merged: DialogueSegment = {
        ...first,
        source_end: second.source_end,
        available_duration: Number((second.source_end - first.source_start).toFixed(2)),
        proposed_dubbed_end: second.proposed_dubbed_end,
        final_approved_end: second.final_approved_end,
        original_text: `${first.original_text} ${second.original_text}`.trim(),
        active_target_text: `${first.active_target_text} ${second.active_target_text}`.trim(),
        natural_spoken_version: `${first.natural_spoken_version} ${second.natural_spoken_version}`.trim(),
        word_events: [...first.word_events, ...second.word_events],
        actual_synthesized_duration: null,
        synthesized_audio_base64: null,
        needs_resynthesis: true
      };

      const next = [...prev.slice(0, idx), merged, ...prev.slice(idx + 2)];
      return evaluateTimelineDurationAlignment(next).updatedSegments;
    });
  };

  const handleAuditionCharacterVoice = async (charId: string) => {

      if (err instanceof ElevenLabsAccountRestrictionError) {

    }
  };

  const playBase64Audio = (base64: string, mimeType: string | null, playbackRate = 1.0) => {
    if (!audioPreviewRef.current) return;
    const raw = base64.trim();
    audioPreviewRef.current.src = raw.startsWith('data:')
      ? raw
      : `data:${mimeType || 'audio/mpeg'};base64,${raw}`;
    audioPreviewRef.current.playbackRate = playbackRate;
    audioPreviewRef.current.play().catch(() => {});
  };

  const handleMergeSpeakerIntoCharacter = (fromCharId: string, targetCharId: string) => {
    if (fromCharId === targetCharId) return;
    const targetChar = characters.find((c) => c.character_id === targetCharId);
    if (!targetChar) return;

    commitSegmentsUpdate((prev) =>
      prev.map((seg) =>
        seg.character_id === fromCharId
          ? { ...seg, character_id: targetCharId, speaker_id: targetChar.original_speaker_id }
          : seg
      )
    );

    setCharacters((prev) => {
      const removed = prev.find((c) => c.character_id === fromCharId);
      return prev
        .filter((c) => c.character_id !== fromCharId)
        .map((c) =>
          c.character_id === targetCharId
            ? { ...c, dialogue_count: c.dialogue_count + (removed?.dialogue_count || 0) }
            : c
        );
    });
  };

  // Real Project Telemetry & QA Counters
  const totalSegs = segments.length;
  const localizedSegsCount = segments.filter(
    (s) => Boolean(s.active_target_text) && !s.needs_retranslation
  ).length;
  const validAudioClipsCount = segments.filter(
    (s) => Boolean(s.synthesized_audio_base64) && !s.needs_resynthesis
  ).length;
  const outdatedAudioClipsCount = segments.filter(
    (s) => Boolean(s.synthesized_audio_base64) && s.needs_resynthesis
  ).length;
  const overflowSegsCount = segments.filter(
    (s) => s.timing_status === TimingStatus.OVERFLOW_FAILED
  ).length;
  const unassignedVoicesCount = characters.filter((c) => !c.assigned_voice_id).length;

  const overallProgressPct =
    totalSegs === 0
      ? 0
      : Math.round(
          ((totalSegs + localizedSegsCount + validAudioClipsCount + (finalValidationReport ? totalSegs : 0)) /
            (totalSegs * 4)) *
            100
        );

  const orderedStages: Array<{
    id: ProductionStageTab;
    stepNum: number;
    label: string;
    status: ConciseStageStatus;
    summary: string;
  }> = [
    {
      id: 'STAGE_1_TRANSCRIBE',
      stepNum: 1,
      label: 'Transcribe',
      status:
        stageProgress.scribe.status === 'running'
          ? 'In Progress'
          : stageProgress.scribe.status === 'failed'
            ? 'Failed'
            : totalSegs > 0
              ? 'Completed'
              : 'Waiting',
      summary: totalSegs > 0 ? `${totalSegs} of ${totalSegs} segments` : 'Import movie'
    },
    {
      id: 'STAGE_2_SPEAKERS',
      stepNum: 2,
      label: 'Verify Speakers',
      status:
        totalSegs === 0
          ? 'Waiting'
          : unassignedVoicesCount > 0 || disagreements.some((d) => !d.resolved)
            ? 'Needs Review'
            : characters.length > 0
              ? 'Completed'
              : 'Waiting',
      summary:
        totalSegs === 0
          ? 'Waiting for transcript'
          : `${characters.length} speakers • ${disagreements.filter((d) => !d.resolved).length} diffs`
    },
    {
      id: 'STAGE_3_LOCALIZE_KHMER',
      stepNum: 3,
      label: 'Localize to Khmer',
      status:
        stageProgress.gemini.status === 'running' || Boolean(translatingSegmentId)

          : stageProgress.gemini.status === 'failed'
            ? 'Failed'
            : totalSegs === 0
              ? 'Waiting'
              : localizedSegsCount === totalSegs
                ? 'Completed'
                : localizedSegsCount > 0
                  ? 'Needs Review'
                  : 'Waiting',
      summary: `${localizedSegsCount} of ${totalSegs} translated`
    },
    {
      id: 'STAGE_4_VOICES_AUDITION',
      stepNum: 4,
      label: 'Generate & Review Voices',
      status:
        stageProgress.synthesis.status === 'running' || Boolean(synthesizingSegmentId)
          ? 'In Progress'
          : stageProgress.synthesis.status === 'failed'
            ? 'Failed'
            : totalSegs === 0
              ? 'Waiting'
              : outdatedAudioClipsCount > 0
                ? 'Needs Review'
                : validAudioClipsCount === totalSegs
                  ? 'Completed'
                  : validAudioClipsCount > 0
                    ? 'Needs Review'
                    : 'Waiting',
      summary: `${validAudioClipsCount} of ${totalSegs} audio clips ready`
    },
    {
      id: 'STAGE_5_SYNCHRONIZE',
      stepNum: 5,
      label: 'Synchronize',
      status:
        overflowSegsCount > 0
          ? 'Needs Review'
          : validAudioClipsCount > 0
            ? 'Completed'
            : 'Waiting',
      summary:
        overflowSegsCount > 0
          ? `${overflowSegsCount} timing overflows`
          : validAudioClipsCount > 0
            ? 'Timing aligned'
            : 'Waiting for audio'

    {
      id: 'STAGE_6_ASSEMBLE_EXPORT',
      stepNum: 6,
      label: 'Assemble & Export',
      status:
        stageProgress.assembly.status === 'running'
          ? 'In Progress'
          : stageProgress.assembly.status === 'failed'

            : finalValidationReport

    segments.find((s) => s.segment_id === selectedSegmentId) ||
    segments[0] ||
    null;

  const selectedCharacter = characters.find((c) => c.character_id === selectedSegment?.character_id);

    <div className="h-screen w-screen flex flex-col bg-[#0B0C10] text-[#F1F5F9] overflow-hidden select-none">

            <Film className="w-4 h-4 text-[#0B0C10]" />

            </div>
            <div className="flex items-center gap-3 text-[11px] text-slate-400">

              <span>•</span>

            className="px-4 py-1.5 rounded-lg bg-[#D4AF37] hover:bg-[#e3be42] text-[#0B0C10] text-xs font-bold flex items-center gap-1.5 shadow"
          >
            <Download className="w-3.5 h-3.5" />
            <span>Export</span>
          </button>
        </div>
      </header>

      <nav className="bg-[#0E1017] border-b border-[#222738] px-5 py-2 flex items-center justify-between gap-2 overflow-x-auto shrink-0">
        <div className="flex items-center gap-1.5">
          {orderedStages.map((st) => {
            const isActive = activeStage === st.id;
            const statusColor =
              st.status === 'Completed'

            className={`px-3 py-1.5 rounded-lg border text-xs font-medium flex items-center gap-1.5 ${

                ? 'bg-[#191E2D] border-[#D4AF37] text-white'

              </span>

          <button
            onClick={() => setShowTimeline(!showTimeline)}
            className={`px-3 py-1.5 rounded-lg border text-xs font-medium flex items-center gap-1.5 ${
              showTimeline
                ? 'bg-[#D4AF37]/15 border-[#D4AF37]/50 text-[#D4AF37]'
                : 'bg-[#12151E] border-[#222738] text-slate-300 hover:text-white'
            }`}
          >
            {showTimeline ? <ChevronDown className="w-3.5 h-3.5" /> : <ChevronUp className="w-3.5 h-3.5" />}
            <span>{showTimeline ? 'Hide Timeline' : 'Show Timeline'}</span>
          </button>
        </div>
      </nav>

      {/* Actionable Error Banner for Gemini Localization */}
      {localizationErrorBanner && (
        <div className="bg-rose-500/15 border-b border-rose-500/40 px-5 py-2 flex flex-wrap items-center justify-between gap-3 text-xs text-rose-200 shrink-0">
          <div className="flex items-center gap-2 min-w-0">
            <AlertTriangle className="w-4 h-4 text-rose-400 shrink-0" />
            <span className="truncate">{localizationErrorBanner}</span>
          </div>
          <div className="flex items-center gap-2 shrink-0">
            <button
              onClick={() => setIsConfigOpen(true)}
              className="px-3 py-1 rounded-lg bg-[#D4AF37] text-[#0B0C10] font-bold flex items-center gap-1"
            >
              <Key className="w-3 h-3" /> Enter Gemini API Key
            </button>
            <button
              onClick={() => handleRunGeminiLocalization()}
              className="px-3 py-1 rounded-lg bg-white/10 hover:bg-white/20 text-white font-semibold"
            >
              Retry Localization
            </button>
            <button
              onClick={() => setLocalizationErrorBanner(null)}
              className="text-slate-400 hover:text-white px-1"
            >
              Dismiss
            </button>
          </div>
        </div>
      )}

      {/* Actionable Error Banner for ElevenLabs API Errors */}
      {elevenLabsErrorBanner && (
        <div className="bg-amber-500/15 border-b border-amber-500/40 px-5 py-2 flex flex-wrap items-center justify-between gap-3 text-xs text-amber-200 shrink-0">
          <div className="flex items-center gap-2 min-w-0">
            <AlertTriangle className="w-4 h-4 text-amber-400 shrink-0" />
            <span className="truncate">{elevenLabsErrorBanner}</span>
          </div>
          <div className="flex items-center gap-2 shrink-0">
            <button
              onClick={() => setIsConfigOpen(true)}
              className="px-3 py-1 rounded-lg bg-amber-400 text-[#0B0C10] font-bold"
            >
              Update ElevenLabs Key
            </button>
            <button
              onClick={() => setElevenLabsErrorBanner(null)}
              className="text-slate-400 hover:text-white px-1"
            >
              Dismiss
            </button>
          </div>
        </div>
      )}

      {/* AREA B: TWO-COLUMN DESKTOP MAIN WORKSPACE */}
      <div className="flex-1 flex flex-col lg:flex-row overflow-hidden">
        {/* LEFT COLUMN: Video Preview Monitor, Subtitle Verification & Audio Source Switcher */}
        <div
          className={`${
            expandVideoPreview ? 'w-full lg:w-7/12' : 'w-full lg:w-5/12'
          } shrink-0 bg-[#10131C] border-r border-[#222738] flex flex-col justify-between overflow-y-auto p-4 space-y-3`}
        >
          <div className="space-y-3">
            {/* Video Player with Subtitle Overlay */}
            <div className="relative rounded-xl overflow-hidden bg-black border border-[#262B3D] aspect-video shadow-lg">
              {sourceVideoUrl ? (
                <video
                  ref={videoRef}
                  src={sourceVideoUrl}
                  className="w-full h-full object-contain"
                  onTimeUpdate={handleVideoTimeUpdate}
                  onLoadedMetadata={() => {
                    if (videoRef.current && Number.isFinite(videoRef.current.duration)) {
                      setVideoDurationSec(Number(videoRef.current.duration.toFixed(2)));
                    }
                  }}
                  onPlay={() => setIsPlaying(true)}
                  onPause={() => setIsPlaying(false)}
                  onEnded={() => setIsPlaying(false)}
                />
              ) : (
                <div className="w-full h-full flex flex-col items-center justify-center p-6 text-center">
                  <FileVideo className="w-8 h-8 text-slate-600 mb-2" />
                  <p className="text-xs text-slate-400 mb-3">
                    {needsVideoReselect
                      ? 'Project checkpoint restored from IndexedDB. Please re-select your source movie file to restore video playback.'
                      : 'Import a movie file (.MP4, .WebM, .MOV, .WAV, .MP3) to start dubbing.'}
                  </p>
                  <label className="px-4 py-2 rounded-lg bg-[#D4AF37] hover:bg-[#e5c247] text-[#0B0C10] text-xs font-bold cursor-pointer flex items-center gap-1.5">
                    <Upload className="w-3.5 h-3.5" />
                    <span>Import Movie File</span>
                    <input
                      type="file"
                      accept="video/*,audio/*"
                      onChange={handleFileUpload}
                      className="hidden"
                    />
                  </label>

                <div

                  }}
                  className={`absolute inset-x-3 ${

                  title="Click subtitle to select this dialogue segment"
