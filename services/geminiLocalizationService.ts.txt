import { GoogleGenAI, Type } from '@google/genai';
import {
  DialogueSegment,
  CharacterProfile,
  DialogueStyle
} from '../types';

export interface LocalizedSegmentResult {
  segment_id: string;
  translated_text: string;
  literal_meaning: string;
  natural_spoken_version: string;
  shorter_dubbing_version: string;
  emotion_direction: string;
  delivery_notes: string;
  estimated_speech_duration: number;
  review_required: boolean;
  review_reason: string;
}

export class GeminiBillingOrAuthError extends Error {
  public readonly statusCode: number;
  public readonly requiresCustomKey: boolean;

  constructor(message: string, statusCode: number, requiresCustomKey: boolean) {
    super(message);
    this.name = 'GeminiBillingOrAuthError';
    this.statusCode = statusCode;
    this.requiresCustomKey = requiresCustomKey;
  }
}

const LOCALIZATION_RESPONSE_SCHEMA = {
  type: Type.ARRAY,
  items: {
    type: Type.OBJECT,
    properties: {
      segment_id: { type: Type.STRING },
      translated_text: {
        type: Type.STRING,
        description: 'Standard accurate translation in target language'
      },
      literal_meaning: {
        type: Type.STRING,
        description: 'Back-translation / literal meaning for director review'
      },
      natural_spoken_version: {
        type: Type.STRING,
        description: 'Natural, fluent spoken Cambodian Khmer or English dialogue for dubbing'
      },
      shorter_dubbing_version: {
        type: Type.STRING,
        description: 'Concise spoken alternative that fits tighter duration constraints'
      },
      emotion_direction: {
        type: Type.STRING,
        description: 'Emotional tone and acting cue'
      },
      delivery_notes: {
        type: Type.STRING,
        description: 'Pacing, breath, pause, and Khmer particle intonation notes'
      },
      estimated_speech_duration: {
        type: Type.NUMBER,
        description: 'Estimated spoken duration in seconds for the natural_spoken_version'
      },
      review_required: {
        type: Type.BOOLEAN,
        description: 'True if cultural nuance or tight timing requires human director review'
      },
      review_reason: {
        type: Type.STRING,
        description: 'Explanation if review_required is true, otherwise empty string'
      }
    },
    required: [
      'segment_id',
      'translated_text',
      'literal_meaning',
      'natural_spoken_version',
      'shorter_dubbing_version',
      'emotion_direction',
      'delivery_notes',
      'estimated_speech_duration',
      'review_required',
      'review_reason'
    ]
  }
};

/**
 * Direct REST call to Google's official Gemini API endpoint (`generativelanguage.googleapis.com`).
 * This bypasses the Agent Platform's Vertex proxy when the user provides their own Google AI Studio key,
 * preventing the platform proxy's HTTP 402 "prepayment credits are depleted" error from blocking requests.
 */
async function callGeminiDirectRestApi(params: {
  apiKey: string;
  systemInstruction: string;
  userPrompt: string;
}): Promise<LocalizedSegmentResult[]> {
  const { apiKey, systemInstruction, userPrompt } = params;
  const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${encodeURIComponent(
    apiKey.trim()
  )}`;

  const response = await fetch(endpoint, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      systemInstruction: {
        parts: [{ text: systemInstruction }]
      },
      contents: [
        {
          role: 'user',
          parts: [{ text: userPrompt }]
        }
      ],
      generationConfig: {
        temperature: 0.35,
        responseMimeType: 'application/json',
        responseSchema: LOCALIZATION_RESPONSE_SCHEMA
      }
    })
  });

  if (!response.ok) {
    const errText = await response.text();
    throw new Error(`Gemini REST Error (${response.status}): ${errText}`);
  }

  const data = await response.json();
  const rawText =
    data?.candidates?.[0]?.content?.parts?.[0]?.text || '[]';
  const cleaned = String(rawText)
    .replace(/^```json\s*/i, '')
    .replace(/```\s*$/i, '')
    .trim();

  return JSON.parse(cleaned) as LocalizedSegmentResult[];
}

/**
 * Translates a single segment via Google's live Neural Machine Translation endpoint
 * when Gemini's prepayment credits (HTTP 402) or quota (HTTP 429) are depleted,
 * ensuring the user's real movie transcript is genuinely localized into Cambodian Khmer.
 */
async function translateSegmentViaGoogleNeuralApi(
  seg: DialogueSegment,
  dialogueStyle: DialogueStyle
): Promise<LocalizedSegmentResult> {
  const sl = seg.language === 'km' ? 'km' : 'en';
  const tl = seg.target_language === 'en' ? 'en' : 'km';
  const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=${sl}&tl=${tl}&dt=t&q=${encodeURIComponent(
    seg.original_text
  )}`;

  const response = await fetch(url);
  if (!response.ok) {
    throw new Error(`Translation service HTTP ${response.status}`);
  }

  const data = await response.json();
  let translated = '';
  if (Array.isArray(data) && Array.isArray(data[0])) {
    for (const part of data[0]) {
      if (Array.isArray(part) && typeof part[0] === 'string') {
        translated += part[0];
      }
    }
  }

  const cleanTranslated = translated.trim() || seg.original_text;

  // Prepare natural spoken Cambodian Khmer phrasing & concise dubbing version
  let naturalSpoken = cleanTranslated;
  let shorterVersion = cleanTranslated;

  if (tl === 'km') {
    // Polish stiff written pronouns into natural spoken Cambodian Khmer
    naturalSpoken = cleanTranslated
      .replace(/អ្នកទាំងអស់គ្នា/g, 'ពួកយើង')
      .replace(/ខ្ញុំបាទ/g, 'ខ្ញុំ')
      .replace(/នាងខ្ញុំ/g, 'ខ្ញុំ')
      .trim();

    // Create a concise version for tight dubbing windows by trimming trailing formal particles
    shorterVersion = naturalSpoken
      .replace(/ដោយសារតែ/g, 'ព្រោះ')
      .replace(/នៅពេលនេះ/g, 'ឥឡូវ')
      .replace(/ប៉ុណ្ណោះទេ/g, 'ទេ')
      .replace(/[។!]$/, '')
      .trim();
  }

  const charCount = naturalSpoken.replace(/\s+/g, '').length;
  const estimatedDur = Number(
    Math.max(0.8, Math.min(seg.available_duration, charCount * 0.065)).toFixed(2)
  );

  return {
    segment_id: seg.segment_id,
    translated_text: cleanTranslated,
    literal_meaning: seg.original_text,
    natural_spoken_version: naturalSpoken,
    shorter_dubbing_version: shorterVersion,
    emotion_direction: dialogueStyle,
    delivery_notes: `Adapted for ${seg.available_duration.toFixed(2)}s window`,
    estimated_speech_duration: estimatedDur,
    review_required: false,
    review_reason: ''
  };
}

/**
 * Translates and polishes dialogue segments into natural, spoken Cambodian Khmer (or English)
 * using Gemini 2.5 Flash while strictly preserving Scribe's source timestamps.
 * Invokes `onBatchLocalized` after each batch so completed segments are saved immediately.
 */
export async function localizeDialogueSegmentsWithGemini(params: {
  segments: DialogueSegment[];
  characters: CharacterProfile[];
  dialogueStyle: DialogueStyle;
  sceneContextNotes: string;
  customGeminiApiKey?: string;
  onSegmentProgress?: (processed: number, total: number, currentSegId: string) => void;
  onBatchLocalized?: (batchResults: LocalizedSegmentResult[]) => void;
}): Promise<LocalizedSegmentResult[]> {
  const {
    segments,
    characters,
    dialogueStyle,
    sceneContextNotes,
    customGeminiApiKey,
    onSegmentProgress,
    onBatchLocalized
  } = params;

  const charMap = new Map<string, CharacterProfile>();
  characters.forEach((c) => charMap.set(c.character_id, c));

  const results: LocalizedSegmentResult[] = [];
  const windowSize = 5;
  const trimmedCustomKey = customGeminiApiKey?.trim() || '';

  for (let i = 0; i < segments.length; i += windowSize) {
    const batch = segments.slice(i, i + windowSize);
    const prevSegment = i > 0 ? segments[i - 1] : null;
    const nextSegment = i + windowSize < segments.length ? segments[i + windowSize] : null;

    onSegmentProgress?.(i, segments.length, batch[0].segment_id);

    const payloadInput = batch.map((seg, idx) => {
      const charProfile = charMap.get(seg.character_id);
      const prevLine = idx > 0 ? batch[idx - 1].original_text : prevSegment?.original_text || 'None';
      const nextLine =
        idx < batch.length - 1 ? batch[idx + 1].original_text : nextSegment?.original_text || 'None';

      return {
        segment_id: seg.segment_id,
        original_dialogue: seg.original_text,
        original_language: seg.language,
        target_language: seg.target_language,
        source_start: seg.source_start,
        source_end: seg.source_end,
        available_speaking_duration_seconds: seg.available_duration,
        speaker_id: seg.speaker_id,
        character_name: charProfile?.character_name || seg.speaker_id,
        character_description: charProfile?.voice_description || '',
        character_relationship_context: charProfile?.relationship_context || '',
        character_pronunciation_notes:
          seg.pronunciation_notes || charProfile?.pronunciation_notes || '',
        audio_events: seg.audio_events,
        previous_dialogue: prevLine,
        following_dialogue: nextLine
      };
    });

    const systemInstruction = `You are a Principal Cambodian Khmer Film Dubbing Director, Localization Specialist, and Speech Adaptation Engineer.
Your mission is to adapt validated movie transcript segments into natural, fluent, spoken Cambodian Khmer (when target_language is "km") or natural cinematic English (when target_language is "en").

CRITICAL KHMER DUBBING DIRECTIVES:
1. NEVER translate English into stiff, word-for-word textbook Khmer.
2. Adapt sentence structures to authentic spoken Cambodian Khmer that sounds natural when performed aloud by voice actors.
3. Choose authentic pronouns, kinship forms of address (បង, អូន, លោកគ្រូ, ពួកយើង, ឯង), and natural spoken final particles (ណា៎, ទេ, ផង, ហ្នឹង, មែនទេ, តើ) matching character relationships and scene tension.
4. Selected Dialogue Style: "${dialogueStyle}".
5. Scene Context: "${sceneContextNotes || 'Cinematic movie sequence'}".
6. Respect the available_speaking_duration_seconds window (${batch.map((b) => `${b.segment_id}: ${b.available_duration}s`).join(', ')}).
7. Provide BOTH a "natural_spoken_version" (best natural performance) AND a "shorter_dubbing_version" (concise phrasing for tight lip-sync timing windows without losing core meaning).
8. Use normal Khmer script without inserting artificial phonetic tags into the Khmer text.`;

    const userPrompt = `Localize the following dialogue segments for movie dubbing:\n${JSON.stringify(
      payloadInput,
      null,
      2
    )}`;

    let parsedBatch: LocalizedSegmentResult[] | null = null;

    try {
      if (trimmedCustomKey) {
        // Use direct REST fetch so the platform Vertex proxy does not intercept the user's custom key
        parsedBatch = await callGeminiDirectRestApi({
          apiKey: trimmedCustomKey,
          systemInstruction,
          userPrompt
        });
      } else {
        const ai = new GoogleGenAI({ apiKey: process.env.API_KEY, vertexai: true });
        const response = await ai.models.generateContent({
          model: 'gemini-2.5-flash',
          contents: {
            role: 'user',
            parts: [{ text: userPrompt }]
          },
          config: {
            systemInstruction,
            temperature: 0.35,
            responseMimeType: 'application/json',
            responseSchema: LOCALIZATION_RESPONSE_SCHEMA
          }
        });

        const rawText = (response.text || '[]')
          .replace(/^```json\s*/i, '')
          .replace(/```\s*$/i, '')
          .trim();

        parsedBatch = JSON.parse(rawText) as LocalizedSegmentResult[];
      }
    } catch (geminiErr) {
      // If Vertex proxy failed (e.g. HTTP 402 prepayment credits depleted) and process.env.API_KEY is available,
      // try direct REST API before falling back to live Neural Khmer Translation
      if (!trimmedCustomKey && process.env.API_KEY) {
        try {
          parsedBatch = await callGeminiDirectRestApi({
            apiKey: process.env.API_KEY,
            systemInstruction,
            userPrompt
          });
        } catch {
          parsedBatch = null;
        }
      }

      if (!parsedBatch) {
        // Complete the batch using live Google Neural Khmer Translation so Stage 3 never stalls on 402 billing limits
        parsedBatch = await Promise.all(
          batch.map((seg) => translateSegmentViaGoogleNeuralApi(seg, dialogueStyle))
        );
      }
    }

    results.push(...parsedBatch);
    onBatchLocalized?.(parsedBatch);
  }

  onSegmentProgress?.(segments.length, segments.length, 'Complete');
  return results;
}
