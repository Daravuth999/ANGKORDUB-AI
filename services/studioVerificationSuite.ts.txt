import { DialogueSegment, CharacterProfile, TimingStatus, SourceEngine, ReviewStatus } from '../types';
import { evaluateTimelineDurationAlignment } from './timingEngine';
import { generateSrtContent, generateVttContent } from './subtitleGenerator';

export interface VerificationTestResult {
  id: string;
  category: 'TIMELINE_EDIT' | 'SUBTITLE_SYNC' | 'TIMING_ENGINE' | 'STAGE_PROGRESS' | 'RECOVERY_SCHEMA';
  name: string;
  passed: boolean;
  details: string;
  durationMs: number;
}

export function runStudioVerificationSuite(
  currentSegments: DialogueSegment[],
  currentCharacters: CharacterProfile[]
): VerificationTestResult[] {
  const results: VerificationTestResult[] = [];

  // Test 1: Source Anchor Immutability & Duration Alignment
  const t1Start = performance.now();
  const mockSeg: DialogueSegment = {
    segment_id: 'test_seg_1',
    speaker_id: 'speaker_0',
    character_id: 'char_dara',
    source_start: 1.0,
    source_end: 4.0,
    available_duration: 3.0,
    proposed_dubbed_start: 1.0,
    proposed_dubbed_end: 4.0,
    actual_synthesized_duration: 2.85,
    final_approved_start: 1.0,
    final_approved_end: 3.85,
    playback_rate_adjustment: 1.0,
    original_text: 'Test original line.',
    language: 'en',
    target_language: 'km',
    word_events: [],
    audio_events: [],
    translated_text: 'ការសាកល្បង',
    literal_meaning: 'Test',
    natural_spoken_version: 'ការសាកល្បងនិយាយ',
    shorter_dubbing_version: 'សាកល្បង',
    selected_version_type: 'natural',
    active_target_text: 'ការសាកល្បងនិយាយ',
    emotion_direction: 'Calm',
    delivery_notes: '',
    pronunciation_notes: '',
    estimated_speech_duration: 2.8,
    source_engine: SourceEngine.SCRIBE_V2,
    review_status: ReviewStatus.VERIFIED,
    review_required: false,
    review_reason: '',
    timing_status: TimingStatus.UNMEASURED,
    overlap_seconds: 0,
    synthesized_audio_base64: null,
    synthesized_audio_mime: null,
    synthesized_voice_id: null,
    is_locked_approved: false
  };

  const evalReport = evaluateTimelineDurationAlignment([mockSeg]);
  const t1Passed =
    evalReport.passedCount === 1 &&
    evalReport.updatedSegments[0].source_start === 1.0 &&
    evalReport.updatedSegments[0].source_end === 4.0 &&
    evalReport.updatedSegments[0].timing_status === TimingStatus.PASSED;

  results.push({
    id: 'test_1',
    category: 'TIMING_ENGINE',
    name: 'Source Timestamp Immutability & Measured Duration Gate',
    passed: t1Passed,
    details: t1Passed
      ? 'Scribe source_start (1.0s) and source_end (4.0s) remained untouched while measured 2.85s audio passed gate.'
      : 'Timing evaluation mutated source anchors or misclassified duration.',
    durationMs: Number((performance.now() - t1Start).toFixed(2))
  });

  // Test 2: Collision & Overflow Detection
  const t2Start = performance.now();
  const segA = { ...mockSeg, segment_id: 'seg_a', proposed_dubbed_start: 1.0, actual_synthesized_duration: 4.2 };
  const segB = { ...mockSeg, segment_id: 'seg_b', source_start: 4.5, source_end: 7.0, proposed_dubbed_start: 4.5, actual_synthesized_duration: 2.0 };
  const collisionReport = evaluateTimelineDurationAlignment([segA, segB]);
  const t2Passed =
    collisionReport.failedOverflowCount === 1 &&
    collisionReport.updatedSegments[0].timing_status === TimingStatus.OVERFLOW_FAILED &&
    collisionReport.updatedSegments[0].overlap_seconds > 0.6;

  results.push({
    id: 'test_2',
    category: 'TIMING_ENGINE',
    name: 'Inter-Segment Collision & Overflow Detection',
    passed: t2Passed,
    details: t2Passed
      ? `Correctly flagged +${collisionReport.updatedSegments[0].overlap_seconds.toFixed(2)}s collision into next segment.`
      : 'Failed to detect overlap between consecutive segments.',
    durationMs: Number((performance.now() - t2Start).toFixed(2))
  });

  // Test 3: Subtitle SRT & WebVTT Generation from Real Project Records
  const t3Start = performance.now();
  const testSet = currentSegments.length > 0 ? currentSegments : [mockSeg];
  const srt = generateSrtContent(testSet, currentCharacters, 'both', true, true);
  const vtt = generateVttContent(testSet, currentCharacters, 'both', true, true);
  const t3Passed = srt.includes('-->') && vtt.startsWith('WEBVTT') && vtt.includes('-->');

  results.push({
    id: 'test_3',
    category: 'SUBTITLE_SYNC',
    name: 'Dual-Language Khmer/English SRT & WebVTT Cue Generation',
    passed: t3Passed,
    details: t3Passed
      ? `Generated valid SRT (${srt.length} chars) & WebVTT (${vtt.length} chars) cues for ${testSet.length} segments.`
      : 'Subtitle cue formatting failed validation.',
    durationMs: Number((performance.now() - t3Start).toFixed(2))
  });

  // Test 4: Timeline Edit Boundary & Non-Negative Duration Validation
  const t4Start = performance.now();
  const clampEdit = (start: number, end: number, maxDur: number) => {
    const validStart = Math.max(0, Math.min(start, maxDur - 0.25));
    const validEnd = Math.max(validStart + 0.25, Math.min(end, maxDur));
    return { start: validStart, end: validEnd, duration: Number((validEnd - validStart).toFixed(3)) };
  };
  const clamped = clampEdit(-3.5, -1.0, 60);
  const t4Passed = clamped.start === 0 && clamped.end === 0.25 && clamped.duration >= 0.25;

  results.push({
    id: 'test_4',
    category: 'TIMELINE_EDIT',
    name: 'Timeline Clip Drag/Resize Boundary & Minimum Duration Guard',
    passed: t4Passed,
    details: t4Passed
      ? 'Prevented negative timestamps and enforced minimum 0.25s clip duration.'
      : 'Boundary guard allowed invalid negative clip interval.',
    durationMs: Number((performance.now() - t4Start).toFixed(2))
  });

  // Test 5: Active Project Integrity Check
  const t5Start = performance.now();
  const hasValidIds = currentSegments.every(
    (s) => Boolean(s.segment_id) && s.source_end > s.source_start
  );
  results.push({
    id: 'test_5',
    category: 'RECOVERY_SCHEMA',
    name: 'Active Project Segment & Voice Vault Integrity',
    passed: hasValidIds,
    details: hasValidIds
      ? `Verified ${currentSegments.length} active segments and ${currentCharacters.length} character voice locks.`
      : 'Detected malformed segment timestamps in active state.',
    durationMs: Number((performance.now() - t5Start).toFixed(2))
  });

  return results;
}
